#!/usr/bin/env python3
"""
Frida script that finds 'detect-fm-<PIN>' processes and captures decrypted
network traffic via three layers plus a diagnostic backtrace logger:

  1. secur32/schannel/sspicli EncryptMessage/DecryptMessage  (sync SSPI)
  2. ws2_32 send/recv/sendto/recvfrom/WSASend/WSARecv        (socket fallback)
  3. ntdll NtDeviceIoControlFile with SSPI IOCTLs            (async SSPI)
  4. Backtrace logger on EncryptMessage (first 20 calls only)

Run elevated (Administrator). Frida 17.x compatible.
"""

import frida
import sys
import json
import time
import re

try:
    from Crypto.Cipher import AES, ARC4
    HAVE_PYCRYPTO = True
except ImportError:
    HAVE_PYCRYPTO = False


JS_AGENT = r"""
'use strict';

const seen = new Set();

function ts() { return new Date().toISOString(); }
function emit(p) { try { send(p); } catch (e) {} }

function bytesToHex(b) {
    let s = '';
    for (let i = 0; i < b.length; i++)
        s += ('0' + (b[i] & 0xff).toString(16)).slice(-2);
    return s;
}

function readMemory(ptr, len) {
    try {
        if (!ptr || ptr.isNull() || len <= 0 || len > 0x100000) return null;
        return new Uint8Array(ptr.readByteArray(len));
    } catch (e) { return null; }
}

function u8ToStr(u8) {
    try {
        let s = '';
        for (let i = 0; i < u8.length; i++) {
            const c = u8[i];
            if (c === 0) break;
            s += String.fromCharCode(c);
        }
        return s;
    } catch (e) { return ''; }
}

function emitEvent(kind, data) {
    emit({ type: 'network_event', kind: kind, timestamp: ts(), data: data });
}

function findExport(moduleName, symbol) {
    try {
        if (typeof Module.getExportByName === 'function') {
            return moduleName
                ? Module.getExportByName(moduleName, symbol)
                : Module.getGlobalExportByName(symbol);
        }
    } catch (e) {}
    try {
        if (typeof Module.findExportByName === 'function') {
            return Module.findExportByName(moduleName, symbol);
        }
    } catch (e) {}
    try {
        if (typeof Module.findGlobalExportByName === 'function') {
            return Module.findGlobalExportByName(symbol);
        }
    } catch (e) {}
    return null;
}

function getModuleSafe(name) {
    try { return Process.getModuleByName(name); } catch (e) { return null; }
}

// Heuristic validator for a PSecBufferDesc
function looksLikeSecBufferDesc(ptr) {
    try {
        if (!ptr || ptr.isNull()) return false;
        const cb = ptr.readU32();
        if (cb < 1 || cb > 16) return false;
        const arr = ptr.add(Process.pointerSize).readPointer();
        if (!arr || arr.isNull()) return false;
        const firstCb   = arr.readU32();
        const firstType = arr.add(4).readU32();
        if (firstCb > 0x100000) return false;
        if (firstType > 0x20) return false;
        return true;
    } catch (e) { return false; }
}

function parseSecBufferDesc(pBuffers, api, direction) {
    try {
        if (!pBuffers || pBuffers.isNull()) return;
        const cBuffers = pBuffers.readU32();
        if (cBuffers <= 0 || cBuffers > 16) return;
        const bufArray = pBuffers.add(Process.pointerSize).readPointer();
        if (!bufArray || bufArray.isNull()) return;

        // x64 SecBuffer: { ULONG cbBuffer; ULONG BufferType; PVOID pvBuffer; }
        // with 8-byte alignment → 16 bytes per entry.
        const stride = 16;

        for (let i = 0; i < cBuffers; i++) {
            const entry = bufArray.add(i * stride);
            let cbBuffer = 0, bufType = 0, pvBuffer = null;
            try {
                cbBuffer = entry.readU32();
                bufType  = entry.add(4).readU32();
                pvBuffer = entry.add(8).readPointer();
            } catch (e) { continue; }

            // 0x01 = SECBUFFER_DATA, 0x07 = SECBUFFER_STREAM
            if (bufType !== 0x01 && bufType !== 0x07) continue;
            if (cbBuffer === 0 || cbBuffer > 0x100000) continue;

            const data = readMemory(pvBuffer, cbBuffer);
            if (!data) continue;

            emitEvent(
                direction === 'out' ? 'outgoing_schannel_plain'
                                    : 'incoming_schannel_plain',
                {
                    api: api,
                    length: cbBuffer,
                    hex: bytesToHex(data),
                    ascii: u8ToStr(data)
                }
            );
        }
    } catch (e) {}
}

// -----------------------------------------------------------------
// Layer 1: sync SSPI + backtrace logger
// -----------------------------------------------------------------
function hookSchannel() {
    if (Process.platform !== 'windows') return;

    const candidates = [
        { mod: 'secur32.dll',  sym: 'EncryptMessage', dir: 'out' },
        { mod: 'secur32.dll',  sym: 'DecryptMessage', dir: 'in'  },
        { mod: 'schannel.dll', sym: 'EncryptMessage', dir: 'out' },
        { mod: 'schannel.dll', sym: 'DecryptMessage', dir: 'in'  },
        { mod: 'sspicli.dll',  sym: 'EncryptMessage', dir: 'out' },
        { mod: 'sspicli.dll',  sym: 'DecryptMessage', dir: 'in'  },
    ];

    const seenAddrs = new Set();
    let btBudget = 20;   // print at most 20 backtraces total

    for (const c of candidates) {
        const addr = findExport(c.mod, c.sym);
        if (!addr) continue;
        const key = addr.toString();
        if (seenAddrs.has(key)) continue;
        seenAddrs.add(key);

        try {
            Interceptor.attach(addr, {
                onEnter(args) {
                    // Backtrace logger — only on EncryptMessage
                    if (c.sym === 'EncryptMessage' && btBudget > 0) {
                        btBudget--;
                        try {
                            const bt = Thread.backtrace(this.context, Backtracer.ACCURATE);
                            const frames = [];
                            for (let j = 0; j < Math.min(bt.length, 12); j++) {
                                try {
                                    const sym = DebugSymbol.fromAddress(bt[j]);
                                    frames.push('#' + j + ' ' + sym);
                                } catch (e) {
                                    frames.push('#' + j + ' ' + bt[j]);
                                }
                            }
                            emit({
                                type: 'backtrace',
                                where: c.mod + '!' + c.sym,
                                frames: frames
                            });
                        } catch (e) {
                            emit({ type: 'backtrace', where: c.mod + '!' + c.sym,
                                   frames: ['<backtrace failed: ' + e + '>'] });
                        }
                    }

                    this.pBuffers = null;
                    for (let i = 0; i < 6; i++) {
                        if (looksLikeSecBufferDesc(args[i])) {
                            this.pBuffers = args[i];
                            break;
                        }
                    }
                },
                onLeave(retval) {
                    if (!this.pBuffers) return;
                    // SEC_E_OK == 0
                    if (retval.toInt32() !== 0) return;
                    parseSecBufferDesc(this.pBuffers, c.mod + '!' + c.sym, c.dir);
                }
            });
            console.log('[*] hooked ' + c.mod + '!' + c.sym);
        } catch (e) {
            console.log('[!] hook ' + c.mod + '!' + c.sym + ' failed: ' + e);
        }
    }
}

// -----------------------------------------------------------------
// Layer 2: ws2_32 socket fallback
// -----------------------------------------------------------------
function hookLibc() {
    if (Process.platform !== 'windows') return;
    const mod = 'ws2_32.dll';
    if (!getModuleSafe(mod)) {
        console.log('[!] ws2_32.dll not loaded');
        return;
    }

    const funcs = [
        ['send',     'out'], ['sendto',   'out'], ['WSASend', 'out'],
        ['recv',     'in'],  ['recvfrom', 'in'],  ['WSARecv', 'in'],
    ];

    for (const [name, dir] of funcs) {
        const addr = findExport(mod, name);
        if (!addr) continue;
        try {
            if (dir === 'out') {
                Interceptor.attach(addr, {
                    onEnter(args) {
                        let buf, len;
                        if (name === 'WSASend') {
                            try {
                                const wsabuf = args[1];
                                len = wsabuf.readU32();
                                buf = wsabuf.add(Process.pointerSize).readPointer();
                            } catch (e) { return; }
                        } else {
                            buf = args[1];
                            len = args[2].toInt32();
                        }
                        if (len <= 0 || len > 0x100000) return;
                        const data = readMemory(buf, len);
                        if (!data) return;
                        emitEvent('outgoing', {
                            api: name, length: len,
                            hex: bytesToHex(data), ascii: u8ToStr(data)
                        });
                    }
                });
            } else {
                Interceptor.attach(addr, {
                    onEnter(args) {
                        this.fd = args[0].toInt32();
                        this.buf = args[1];
                        this.api = name;
                    },
                    onLeave(retval) {
                        const n = retval.toInt32();
                        if (n <= 0 || n > 0x100000) return;
                        let data = null;
                        if (this.api === 'WSARecv') {
                            try {
                                const wsabuf = this.buf;
                                const realLen = wsabuf.readU32();
                                const realBuf = wsabuf.add(Process.pointerSize).readPointer();
                                data = readMemory(realBuf, realLen);
                            } catch (e) {}
                        } else {
                            data = readMemory(this.buf, n);
                        }
                        if (!data) return;
                        emitEvent('incoming', {
                            api: this.api, fd: this.fd, length: n,
                            hex: bytesToHex(data), ascii: u8ToStr(data)
                        });
                    }
                });
            }
            console.log('[*] hooked ' + mod + '!' + name);
        } catch (e) {
            console.log('[!] hook ' + mod + '!' + name + ' failed: ' + e);
        }
    }
}

// -----------------------------------------------------------------
// Layer 3: NtDeviceIoControlFile — async SSPI via lsass
// -----------------------------------------------------------------
function hookNtIoControl() {
    if (Process.platform !== 'windows') return;
    const addr = findExport('ntdll.dll', 'NtDeviceIoControlFile');
    if (!addr) {
        console.log('[!] ntdll!NtDeviceIoControlFile not found');
        return;
    }

    const IOCTL_MAP = {
        0x39000c: 'SSPI_ENCRYPT',
        0x390010: 'SSPI_DECRYPT',
        0x390014: 'SSPI_MAKE_SIGNATURE',
        0x390018: 'SSPI_VERIFY_SIGNATURE',
        0x390004: 'SSPI_CONTEXT',
    };

    try {
        Interceptor.attach(addr, {
            onEnter(args) {
                try {
                    const raw = args[5];
                    const ioctl = raw.toInt32() & 0xffffff;
                    const label = IOCTL_MAP[ioctl];
                    if (!label) return;

                    const inBuf  = args[6];
                    const inLen  = args[7].toInt32();
                    const outBuf = args[8];
                    const outLen = args[9].toInt32();

                    if ((label === 'SSPI_ENCRYPT' || label === 'SSPI_MAKE_SIGNATURE')
                        && inBuf && !inBuf.isNull() && inLen > 0 && inLen < 0x100000) {
                        const data = readMemory(inBuf, inLen);
                        if (data) {
                            emitEvent('outgoing_schannel_plain', {
                                api: 'NtDeviceIoControlFile!' + label,
                                length: inLen,
                                hex: bytesToHex(data),
                                ascii: u8ToStr(data)
                            });
                        }
                    }

                    if ((label === 'SSPI_DECRYPT' || label === 'SSPI_VERIFY_SIGNATURE')
                        && outBuf && !outBuf.isNull() && outLen > 0 && outLen < 0x100000) {
                        this.outBuf = outBuf;
                        this.outLen = outLen;
                        this.label  = label;
                    }
                } catch (e) {}
            },
            onLeave(retval) {
                if (!this.outBuf) return;
                try {
                    const data = readMemory(this.outBuf, this.outLen);
                    if (data) {
                        emitEvent('incoming_schannel_plain', {
                            api: 'NtDeviceIoControlFile!' + this.label,
                            length: this.outLen,
                            hex: bytesToHex(data),
                            ascii: u8ToStr(data)
                        });
                    }
                } catch (e) {}
            }
        });
        console.log('[*] hooked ntdll!NtDeviceIoControlFile');
    } catch (e) {
        console.log('[!] NtDeviceIoControlFile hook failed: ' + e);
    }
}

// -----------------------------------------------------------------
// Entry
// -----------------------------------------------------------------
setImmediate(function () {
    console.log('[*] AGENT VERSION v4 loaded into ' + Process.platform
                + ' pid ' + Process.id + ' (' + Process.arch + ')');
    try { hookSchannel();    } catch (e) { console.log('[!] hookSchannel: ' + e); }
    try { hookLibc();        } catch (e) { console.log('[!] hookLibc: ' + e); }
    try { hookNtIoControl(); } catch (e) { console.log('[!] hookNtIoControl: ' + e); }
    console.log('[*] hooks installed');
});
"""


# ---------------------------------------------------------------------------
# Python side
# ---------------------------------------------------------------------------
HEX_RE = re.compile(r'^[0-9a-fA-F]+$')

TLS_PREFIXES = ('160301', '160302', '160303',
                '140301', '140303',
                '170301', '170303',
                '150301', '150303')


def looks_like_json(s):
    s = (s or '').strip()
    if not s or s[0] not in '{[':
        return False
    try:
        json.loads(s)
        return True
    except Exception:
        return False


def is_printable(b, threshold=0.85):
    if not b:
        return False
    printable = sum(1 for c in b if 32 <= c < 127 or c in (9, 10, 13))
    return (printable / len(b)) >= threshold


def get_device():
    return frida.get_local_device()


def find_detect_processes(device=None):
    if device is None:
        device = get_device()
    matches = []
    try:
        for proc in device.enumerate_processes():
            name = (getattr(proc, 'name', '') or '').lower()
            if 'detect-fm' in name:
                matches.append(proc)
    except Exception as e:
        print(f'[!] enumerate_processes failed: {e}')
    return matches


def on_message(message, data):
    if message.get('type') == 'send':
        payload = message.get('payload')
        if not isinstance(payload, dict):
            return
        ptype = payload.get('type')
        if ptype == 'network_event':
            handle_network_event(payload)
        elif ptype == 'backtrace':
            print('\n[BACKTRACE] ' + payload.get('where', '?'))
            for frame in payload.get('frames', []):
                print('  ' + frame)
    elif message.get('type') == 'error':
        print('[frida error] ' + message.get('description', ''))


def handle_network_event(evt):
    kind = evt.get('kind', '')
    data = evt.get('data', {})
    t = evt.get('timestamp', '')
    ascii_str = data.get('ascii', '') or ''
    hex_str = data.get('hex', '')
    api = data.get('api', '?')
    length = data.get('length', '?')

    keep_all = ('outgoing_schannel_plain', 'incoming_schannel_plain',
                'outgoing_winhttp', 'incoming_winhttp')

    if kind not in keep_all:
        sample = ascii_str[:32]
        if len(sample) < 4:
            return
        printable = sum(1 for c in sample if 32 <= ord(c) < 127)
        if printable / len(sample) < 0.7:
            return
        if hex_str[:6] in TLS_PREFIXES:
            return

    print('\n' + '=' * 78)
    print(f'[{t}] {kind.upper():<26} {api}  len={length}')
    print('-' * 78)

    if ascii_str:
        print('ASCII: ' + repr(ascii_str[:400]))

    if looks_like_json(ascii_str):
        try:
            print('JSON:')
            print(json.dumps(json.loads(ascii_str), indent=2)[:4000])
        except Exception:
            pass

    if hex_str and HEX_RE.match(hex_str) and HAVE_PYCRYPTO:
        try:
            raw = bytes.fromhex(hex_str)
        except Exception:
            raw = b''
        if raw and not is_printable(raw):
            print('[*] ciphertext detected - attempting common-key decryption...')
            for k in (b'0123456789abcdef', b'passwordpassword', b'\x00' * 16):
                try:
                    c = AES.new(k, AES.MODE_ECB)
                    pt = c.decrypt(raw[:len(raw) - (len(raw) % 16)])
                    if is_printable(pt):
                        print('[+] AES-ECB key=' + k.hex() + ':')
                        print('    ' + pt.decode('utf-8', 'replace')[:1500])
                        break
                except Exception:
                    pass


def attach_and_monitor(device, proc, session_holder):
    print(f'\n[+] Attaching to {proc.name} (pid={proc.pid})')
    try:
        session = device.attach(proc.pid)
    except Exception as e:
        print(f'[!] attach failed: {e}')
        return None

    script = session.create_script(JS_AGENT)
    script.on('message', on_message)
    try:
        script.load()
    except Exception as e:
        print(f'[!] script.load failed: {e}')
        try: session.detach()
        except Exception: pass
        return None

    session_holder.append((proc, session, script))
    print(f'[+] Instrumented pid={proc.pid}')
    return session


def main():
    print('[*] Acquiring device...')
    device = get_device()
    print(f'[*] Using device: {device}')

    print('[*] Searching for processes matching "detect-fm"...')
    matches = find_detect_processes(device)
    for p in matches:
        print(f'    - {p.name} (pid={p.pid})')

    sessions = []
    for p in matches:
        attach_and_monitor(device, p, sessions)

    if not sessions:
        print('[*] Polling every 2s (Ctrl+C to quit)...')
        known = set()
        try:
            while True:
                for p in find_detect_processes(device):
                    if p.pid not in known:
                        known.add(p.pid)
                        attach_and_monitor(device, p, sessions)
                time.sleep(2)
        except KeyboardInterrupt:
            print('\n[*] Stopping...')
    else:
        print('\n[*] Monitoring. Press Ctrl+C to exit.')
        try:
            while True:
                time.sleep(1)
        except KeyboardInterrupt:
            pass

    for proc, sess, script in sessions:
        try: script.unload()
        except Exception: pass
        try: sess.detach()
        except Exception: pass


if __name__ == '__main__':
    main()
