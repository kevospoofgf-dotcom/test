#!/usr/bin/env python3
"""
Frida script to find 'detect-fm-<PIN>' processes and monitor network traffic.
Hooks Windows Schannel (EncryptMessage/DecryptMessage) to capture PLAINTEXT
before TLS encryption and after TLS decryption.
Compatible with Frida 17.x on Windows / Linux / macOS.
"""

import frida
import sys
import json
import time
import re
import base64

try:
    from Crypto.Cipher import AES, ARC4
    from Crypto.Util.Padding import unpad
    HAVE_PYCRYPTO = True
except ImportError:
    HAVE_PYCRYPTO = False
    print("[!] pycryptodome not installed - AES/RC4 decryption disabled")
    print("    Install with: python -m pip install pycryptodome")


# ---------------------------------------------------------------------------
# Frida agent
# ---------------------------------------------------------------------------
JS_AGENT = r"""
'use strict';

const seen = new Set();

function ts() { return new Date().toISOString(); }

function emit(payload) {
    try { send(payload); } catch (e) { console.log('[!] send failed: ' + e); }
}

function bytesToHex(bytes) {
    let s = '';
    for (let i = 0; i < bytes.length; i++) {
        s += ('0' + (bytes[i] & 0xff).toString(16)).slice(-2);
    }
    return s;
}

function readMemory(ptr, len) {
    try {
        if (!ptr || ptr.isNull() || len <= 0 || len > 0x100000) return null;
        return new Uint8Array(ptr.readByteArray(len));
    } catch (e) { return null; }
}

function u8ToStr(u8) {
    try {
        let s = '';
        for (let i = 0; i < u8.length; i++) {
            const c = u8[i];
            if (c === 0) break;
            s += String.fromCharCode(c);
        }
        return s;
    } catch (e) { return ''; }
}

function emitEvent(kind, data) {
    emit({ type: 'network_event', kind: kind, timestamp: ts(), data: data });
}

// Frida 17 removed findExportByName in favor of getExportByName.
function findExport(moduleName, symbol) {
    try {
        if (typeof Module.getExportByName === 'function') {
            return moduleName
                ? Module.getExportByName(moduleName, symbol)
                : Module.getGlobalExportByName(symbol);
        }
    } catch (e) {}
    try {
        if (typeof Module.findExportByName === 'function') {
            return Module.findExportByName(moduleName, symbol);
        }
    } catch (e) {}
    try {
        if (typeof Module.findGlobalExportByName === 'function') {
            return Module.findGlobalExportByName(symbol);
        }
    } catch (e) {}
    return null;
}

function getModuleSafe(name) {
    try { return Process.getModuleByName(name); } catch (e) { return null; }
}

// -----------------------------------------------------------------
// Module dump (diagnostic)
// -----------------------------------------------------------------
function dumpModules() {
    const mods = Process.enumerateModules();
    console.log('=== TLS-RELATED MODULES (' + mods.length + ' total) ===');
    for (const m of mods) {
        const n = m.name.toLowerCase();
        if (n.includes('ssl') || n.includes('tls') || n.includes('crypto')
            || n.includes('curl') || n.includes('nss') || n.includes('schannel')
            || n.includes('secur32') || n.includes('sspicli') || n.includes('ncrypt')
            || n.includes('boring') || n.includes('mbed') || n.includes('wolf')) {
            console.log('  ' + m.name + ' @ ' + m.base + ' (' + m.size + ' bytes)');
        }
    }
}

// -----------------------------------------------------------------
// Windows Schannel hooks — plaintext before/after TLS
// -----------------------------------------------------------------
function hookSchannel() {
    if (Process.platform !== 'windows') return;

    const candidates = [
        { mod: 'secur32.dll', sym: 'EncryptMessage' },
        { mod: 'secur32.dll', sym: 'DecryptMessage' },
        { mod: 'schannel.dll', sym: 'EncryptMessage' },
        { mod: 'schannel.dll', sym: 'DecryptMessage' },
        { mod: 'sspicli.dll', sym: 'EncryptMessage' },
        { mod: 'sspicli.dll', sym: 'DecryptMessage' },
    ];

    const seenAddrs = new Set();

    for (const c of candidates) {
        const addr = findExport(c.mod, c.sym);
        if (!addr) continue;
        const key = addr.toString();
        if (seenAddrs.has(key)) continue;
        seenAddrs.add(key);

        const isEncrypt = c.sym.indexOf('Encrypt') !== -1;

        try {
            if (isEncrypt) {
                Interceptor.attach(addr, {
                    onEnter(args) {
                        // EncryptMessage(PSecHandle, ULONG, PSecBufferDesc, ULONG)
                        try {
                            const pBuffers = args[2];
                            const cBuffers = pBuffers.readU32();
                            const bufArray = pBuffers.add(Process.pointerSize).readPointer();
                            for (let i = 0; i < cBuffers; i++) {
                                const entry = bufArray.add(i * 3 * Process.pointerSize);
                                const cbBuffer = entry.readU32();
                                const bufType = entry.add(8).readU32();
                                const pvBuffer = entry.add(16).readPointer();
                                // 0x01 = SECBUFFER_DATA
                                if (bufType !== 0x01) continue;
                                if (cbBuffer === 0 || cbBuffer > 0x100000) continue;
                                const data = readMemory(pvBuffer, cbBuffer);
                                if (!data) continue;
                                emitEvent('outgoing_schannel_plain', {
                                    api: c.mod + '!' + c.sym,
                                    length: cbBuffer,
                                    hex: bytesToHex(data),
                                    ascii: u8ToStr(data)
                                });
                            }
                        } catch (e) {}
                    }
                });
            } else {
                Interceptor.attach(addr, {
                    onEnter(args) {
                        this.pBuffers = args[2];
                    },
                    onLeave(retval) {
                        // SEC_E_OK == 0
                        if (retval.toInt32() !== 0) return;
                        try {
                            const pBuffers = this.pBuffers;
                            const cBuffers = pBuffers.readU32();
                            const bufArray = pBuffers.add(Process.pointerSize).readPointer();
                            for (let i = 0; i < cBuffers; i++) {
                                const entry = bufArray.add(i * 3 * Process.pointerSize);
                                const cbBuffer = entry.readU32();
                                const bufType = entry.add(8).readU32();
                                const pvBuffer = entry.add(16).readPointer();
                                if (bufType !== 0x01) continue;
                                if (cbBuffer === 0 || cbBuffer > 0x100000) continue;
                                const data = readMemory(pvBuffer, cbBuffer);
                                if (!data) continue;
                                emitEvent('incoming_schannel_plain', {
                                    api: c.mod + '!' + c.sym,
                                    length: cbBuffer,
                                    hex: bytesToHex(data),
                                    ascii: u8ToStr(data)
                                });
                            }
                        } catch (e) {}
                    }
                });
            }
            console.log('[*] hooked ' + c.mod + '!' + c.sym + ' @ ' + addr);
        } catch (e) {
            console.log('[!] failed to hook ' + c.mod + '!' + c.sym + ': ' + e);
        }
    }
}

// -----------------------------------------------------------------
// libc / ws2_32 — fallback socket-level capture
// -----------------------------------------------------------------
function hookLibc() {
    let libcName;
    if (Process.platform === 'windows') libcName = 'ws2_32.dll';
    else if (Process.platform === 'darwin') libcName = 'libsystem_c.dylib';
    else libcName = 'libc.so.6';

    let libc = getModuleSafe(libcName);
    if (!libc && Process.platform === 'linux') libc = getModuleSafe('libc.so');
    if (!libc) {
        console.log('[!] could not find libc-like module (' + libcName + ')');
        return;
    }

    const funcs = Process.platform === 'windows'
        ? ['send', 'recv', 'sendto', 'recvfrom', 'WSASend', 'WSARecv']
        : ['send', 'recv', 'sendto', 'recvfrom', 'write', 'read'];

    for (const name of funcs) {
        let addr = findExport(libcName, name);
        if (!addr) addr = findExport(null, name);
        if (!addr) continue;

        const isSend = (name === 'send' || name === 'sendto'
                        || name === 'write' || name === 'WSASend');

        try {
            if (isSend) {
                Interceptor.attach(addr, {
                    onEnter(args) {
                        const fd = args[0].toInt32();
                        if (name === 'WSASend') {
                            try {
                                const wsabuf = args[1];
                                const realLen = wsabuf.readU32();
                                const realBuf = wsabuf.add(Process.pointerSize).readPointer();
                                const data = readMemory(realBuf, realLen);
                                if (data) {
                                    emitEvent('outgoing', {
                                        api: name, fd: fd, length: realLen,
                                        hex: bytesToHex(data), ascii: u8ToStr(data)
                                    });
                                }
                            } catch (e) {}
                            return;
                        }
                        const buf = args[1];
                        const len = args[2].toInt32();
                        if (len <= 0 || len > 0x100000) return;
                        const data = readMemory(buf, len);
                        if (!data) return;
                        emitEvent('outgoing', {
                            api: name, fd: fd, length: len,
                            hex: bytesToHex(data), ascii: u8ToStr(data)
                        });
                    }
                });
            } else {
                Interceptor.attach(addr, {
                    onEnter(args) {
                        this.fd = args[0].toInt32();
                        this.buf = args[1];
                        this.api = name;
                    },
                    onLeave(retval) {
                        const n = retval.toInt32();
                        if (n <= 0 || n > 0x100000) return;
                        let data = null;
                        if (this.api === 'WSARecv') {
                            try {
                                const wsabuf = this.buf;
                                const realLen = wsabuf.readU32();
                                const realBuf = wsabuf.add(Process.pointerSize).readPointer();
                                data = readMemory(realBuf, realLen);
                            } catch (e) {}
                        } else {
                            data = readMemory(this.buf, n);
                        }
                        if (!data) return;
                        emitEvent('incoming', {
                            api: this.api, fd: this.fd, length: n,
                            hex: bytesToHex(data), ascii: u8ToStr(data)
                        });
                    }
                });
            }
            console.log('[*] hooked ' + libcName + '!' + name);
        } catch (e) {
            console.log('[!] failed to hook ' + name + ': ' + e);
        }
    }
}

// -----------------------------------------------------------------
// OpenSSL / NSS / TLS (for non-Schannel targets)
// -----------------------------------------------------------------
function hookOpenSSL() {
    const symbols = [
        'SSL_write', 'SSL_read', 'SSL_write_ex', 'SSL_read_ex',
        'SSL_Write', 'SSL_Read',
        'EVP_EncryptUpdate', 'EVP_DecryptUpdate'
    ];
    const modules = Process.enumerateModules();
    const targets = [];

    for (const m of modules) {
        const n = m.name.toLowerCase();
        if (!(n.includes('ssl') || n.includes('crypto') || n.includes('tls')
              || n.includes('curl') || n.includes('nspr') || n.includes('nss')
              || n.includes('boring') || n.includes('mbed') || n.includes('wolf'))) {
            continue;
        }
        for (const sym of symbols) {
            const a = findExport(m.name, sym);
            if (a) targets.push({ sym: sym, addr: a, module: m.name });
        }
    }

    for (const t of targets) {
        const key = t.module + '!' + t.sym;
        if (seen.has(key)) continue;
        seen.add(key);

        const isWrite = t.sym.toLowerCase().indexOf('write') !== -1
                     || t.sym.toLowerCase().indexOf('encrypt') !== -1;
        const isEx = t.sym.endsWith('_ex');

        try {
            if (isEx) {
                Interceptor.attach(t.addr, {
                    onEnter(args) {
                        this.buf = args[1];
                        this.lenPtr = args[2];
                    },
                    onLeave(retval) {
                        try {
                            if (retval.toInt32() !== 1) return;
                            const n = this.lenPtr.readULong();
                            if (n <= 0 || n > 0x100000) return;
                            const data = readMemory(this.buf, n);
                            if (!data) return;
                            emitEvent(isWrite ? 'outgoing_tls' : 'incoming_tls', {
                                api: key, length: n,
                                hex: bytesToHex(data), ascii: u8ToStr(data)
                            });
                        } catch (e) {}
                    }
                });
            } else {
                Interceptor.attach(t.addr, {
                    onEnter(args) {
                        this.buf = args[1];
                    },
                    onLeave(retval) {
                        const n = retval.toInt32();
                        if (n <= 0 || n > 0x100000) return;
                        const data = readMemory(this.buf, n);
                        if (!data) return;
                        emitEvent(isWrite ? 'outgoing_tls' : 'incoming_tls', {
                            api: key, length: n,
                            hex: bytesToHex(data), ascii: u8ToStr(data)
                        });
                    }
                });
            }
            console.log('[*] hooked ' + key);
        } catch (e) {}
    }
}

// -----------------------------------------------------------------
// Java / Android
// -----------------------------------------------------------------
function hookJava() {
    if (typeof Java === 'undefined' || !Java.available) return;
    Java.perform(function () {
        try {
            const ResponseBody = Java.use('okhttp3.ResponseBody');
            ResponseBody.bytes.implementation = function () {
                const b = this.bytes();
                try {
                    emitEvent('http_response_java', {
                        api: 'okhttp3.ResponseBody.bytes',
                        length: b.length,
                        ascii: Java.array_to_string ? Java.array_to_string(b) : null
                    });
                } catch (e) {}
                return b;
            };
        } catch (e) {}

        try {
            const OutputStream = Java.use('java.io.OutputStream');
            OutputStream.write.overload('[B').implementation = function (b) {
                try {
                    emitEvent('java_out', {
                        api: this.getClass().getName() + '.write',
                        length: b.length,
                        ascii: Java.array_to_string ? Java.array_to_string(b) : ''
                    });
                } catch (e) {}
                return this.write(b);
            };
        } catch (e) {}
    });
}

// -----------------------------------------------------------------
// Windows WinHTTP / WinINet
// -----------------------------------------------------------------
function hookWindowsHttp() {
    if (Process.platform !== 'windows') return;

    const mods = ['winhttp.dll', 'wininet.dll'];
    for (const modName of mods) {
        if (!getModuleSafe(modName)) continue;

        const syms = modName === 'winhttp.dll'
            ? ['WinHttpWriteData', 'WinHttpReadData']
            : ['InternetWriteFile', 'InternetReadFile'];

        for (const sym of syms) {
            const addr = findExport(modName, sym);
            if (!addr) continue;

            const isWrite = sym.toLowerCase().includes('write');

            try {
                if (isWrite) {
                    Interceptor.attach(addr, {
                        onEnter(args) {
                            const buf = args[1];
                            const len = args[2].toInt32();
                            if (len <= 0 || len > 0x100000) return;
                            const data = readMemory(buf, len);
                            if (!data) return;
                            emitEvent('outgoing_winhttp', {
                                api: modName + '!' + sym, length: len,
                                hex: bytesToHex(data), ascii: u8ToStr(data)
                            });
                        }
                    });
                } else {
                    Interceptor.attach(addr, {
                        onEnter(args) {
                            this.buf = args[1];
                            this.lenPtr = args[2];
                            this.api = sym;
                        },
                        onLeave(retval) {
                            try {
                                const n = this.lenPtr.readU32();
                                if (n <= 0 || n > 0x100000) return;
                                const data = readMemory(this.buf, n);
                                if (!data) return;
                                emitEvent('incoming_winhttp', {
                                    api: modName + '!' + this.api, length: n,
                                    hex: bytesToHex(data), ascii: u8ToStr(data)
                                });
                            } catch (e) {}
                        }
                    });
                }
                console.log('[*] hooked ' + modName + '!' + sym);
            } catch (e) {}
        }
    }
}

// -----------------------------------------------------------------
// Entry
// -----------------------------------------------------------------
setImmediate(function () {
    console.log('[*] detect_monitor loaded into ' + Process.platform
                + ' pid ' + Process.id + ' (' + Process.arch + ')');
    try { dumpModules(); } catch (e) { console.log('[!] dumpModules: ' + e); }
    try { hookSchannel(); } catch (e) { console.log('[!] hookSchannel: ' + e); }
    try { hookLibc(); } catch (e) { console.log('[!] hookLibc: ' + e); }
    try { hookOpenSSL(); } catch (e) { console.log('[!] hookOpenSSL: ' + e); }
    try { hookWindowsHttp(); } catch (e) { console.log('[!] hookWindowsHttp: ' + e); }
    try { hookJava(); } catch (e) { console.log('[!] hookJava: ' + e); }
    console.log('[*] hooks installed');
});
"""


# ---------------------------------------------------------------------------
# Decryption helpers
# ---------------------------------------------------------------------------
HEX_RE = re.compile(r'^[0-9a-fA-F]+$')

# TLS record type + version prefixes we should silently drop
TLS_PREFIXES = ('160301', '160303', '160302', '140301', '140303',
                '170301', '170303', '150303', '150301')


def looks_like_json(s):
    s = (s or '').strip()
    if not s or s[0] not in '{[':
        return False
    try:
        json.loads(s)
        return True
    except Exception:
        return False


def is_printable(b, threshold=0.85):
    if not b:
        return False
    printable = sum(1 for c in b if 32 <= c < 127 or c in (9, 10, 13))
    return (printable / len(b)) >= threshold


def try_decrypt_blob(data):
    results = []
    if not HAVE_PYCRYPTO:
        return results

    common_keys = [
        b'0123456789abcdef', b'0123456789ABCDEF',
        b'0000000000000000', b'\x00' * 16,
        b'1234567890123456', b'passwordpassword',
    ]
    iv_zero = b'\x00' * 16

    for k in common_keys:
        try:
            c = AES.new(k, AES.MODE_ECB)
            pt = c.decrypt(data[:len(data) - (len(data) % 16)])
            try: pt = unpad(pt, 16)
            except Exception: pass
            if is_printable(pt):
                results.append(('AES-ECB key=' + k.hex(), pt))
        except Exception:
            pass

        try:
            c = AES.new(k, AES.MODE_CBC, iv_zero)
            pt = c.decrypt(data[:len(data) - (len(data) % 16)])
            try: pt = unpad(pt, 16)
            except Exception: pass
            if is_printable(pt):
                results.append(('AES-CBC zeroIV key=' + k.hex(), pt))
        except Exception:
            pass

    for k in [b'key', b'secret', b'password']:
        try:
            pt = ARC4.new(k).decrypt(data)
            if is_printable(pt):
                results.append(('RC4 key=' + k.hex(), pt))
        except Exception:
            pass

    for key in range(1, 256):
        pt = bytes(b ^ key for b in data[:512])
        if looks_like_json(pt.decode('utf-8', errors='ignore')):
            results.append(('XOR key=0x%02x' % key, pt))
            break

    try:
        s = data.decode('ascii', errors='ignore')
        if re.fullmatch(r'[A-Za-z0-9+/=\s]+', s) and len(s) % 4 == 0:
            dec = base64.b64decode(s)
            if is_printable(dec):
                results.append(('Base64', dec))
    except Exception:
        pass

    return results


# ---------------------------------------------------------------------------
# Process discovery
# ---------------------------------------------------------------------------
def get_device():
    return frida.get_local_device()
    # Android over USB:  return frida.get_usb_device(timeout=5)


def find_detect_processes(device=None):
    if device is None:
        device = get_device()

    matches = []
    try:
        procs = device.enumerate_processes()
    except Exception as e:
        print(f'[!] enumerate_processes failed: {e}')
        return matches

    for proc in procs:
        name = (getattr(proc, 'name', '') or '').lower()
        if 'detect-fm' in name:
            matches.append(proc)
    return matches


# ---------------------------------------------------------------------------
# Event handling
# ---------------------------------------------------------------------------
def on_message(message, data):
    if message.get('type') == 'send':
        payload = message.get('payload')
        if not isinstance(payload, dict):
            print('[raw] ' + str(payload))
            return
        if payload.get('type') == 'network_event':
            handle_network_event(payload)
        else:
            print(json.dumps(payload, indent=2))
    elif message.get('type') == 'error':
        print('[frida error] ' + message.get('description', ''))
        stack = message.get('stack', '')
        if stack:
            print(stack)


def handle_network_event(evt):
    kind = evt.get('kind', '')
    data = evt.get('data', {})
    t = evt.get('timestamp', '')

    api = data.get('api', '?')
    fd = data.get('fd', '?')
    length = data.get('length', '?')
    ascii_preview = (data.get('ascii', '') or '')[:400]
    hex_str = data.get('hex', '')

    # Filter out raw socket-level TLS framing noise (encrypted records).
    # We only want Schannel plaintext events and any non-TLS socket traffic.
    if kind in ('incoming', 'outgoing'):
        if hex_str[:6] in TLS_PREFIXES:
            return
        if length in (1, 9):  # tiny heartbeat/keepalive packets
            return

    print('\n' + '=' * 78)
    print(f'[{t}] {kind.upper():<24} {api}  fd={fd}  len={length}')
    print('-' * 78)

    if ascii_preview:
        print('ASCII: ' + repr(ascii_preview))

    if looks_like_json(data.get('ascii', '')):
        try:
            print('JSON (plaintext):')
            print(json.dumps(json.loads(data.get('ascii', '')), indent=2)[:4000])
        except Exception:
            pass

    if hex_str and HEX_RE.match(hex_str):
        try:
            raw = bytes.fromhex(hex_str)
        except Exception:
            raw = b''

        if raw:
            if not is_printable(raw):
                print('[*] ciphertext (not printable) - attempting decryption...')
                results = try_decrypt_blob(raw)
                if results:
                    for scheme, pt in results:
                        print(f'[+] DECRYPTED via {scheme}:')
                        try:
                            txt = pt.decode('utf-8', errors='replace')
                        except Exception:
                            txt = repr(pt)
                        print('    ' + txt[:2000].replace('\n', '\n    '))
                        if looks_like_json(txt):
                            try:
                                print('    JSON:')
                                print('    ' + json.dumps(json.loads(txt), indent=2)
                                      [:2000].replace('\n', '\n    '))
                            except Exception:
                                pass
                else:
                    print('[*] no decryption succeeded with common schemes')
            else:
                text = raw.decode('utf-8', errors='replace')
                if looks_like_json(text):
                    try:
                        print('JSON:')
                        print(json.dumps(json.loads(text), indent=2)[:4000])
                    except Exception:
                        pass


# ---------------------------------------------------------------------------
# Attach + monitor
# ---------------------------------------------------------------------------
def attach_and_monitor(device, proc, session_holder):
    print(f'\n[+] Attaching to {proc.name} (pid={proc.pid})')
    try:
        session = device.attach(proc.pid)
    except Exception as e:
        print(f'[!] attach failed: {e}')
        return None

    script = session.create_script(JS_AGENT)
    script.on('message', on_message)
    try:
        script.load()
    except Exception as e:
        print(f'[!] script.load failed: {e}')
        try: session.detach()
        except Exception: pass
        return None

    session_holder.append((proc, session, script))
    print(f'[+] Instrumented pid={proc.pid}')
    return session


# ---------------------------------------------------------------------------
# Main
# ---------------------------------------------------------------------------
def main():
    print('[*] Acquiring device...')
    device = get_device()
    print(f'[*] Using device: {device}')

    print('[*] Searching for processes matching "detect-fm"...')
    matches = find_detect_processes(device)

    if matches:
        for p in matches:
            print(f'    - {p.name} (pid={p.pid})')
    else:
        print('[!] No matching processes found. Will keep polling.')

    sessions = []
    for p in matches:
        attach_and_monitor(device, p, sessions)

    if not sessions:
        print('[*] Polling every 2s for new "detect-fm" processes (Ctrl+C to quit)...')
        known = set()
        try:
            while True:
                for p in find_detect_processes(device):
                    if p.pid not in known:
                        known.add(p.pid)
                        attach_and_monitor(device, p, sessions)
                time.sleep(2)
        except KeyboardInterrupt:
            print('\n[*] Stopping...')
    else:
        print('\n[*] Monitoring. Press Ctrl+C to exit.')
        try:
            while True:
                time.sleep(1)
        except KeyboardInterrupt:
            pass

    for proc, sess, script in sessions:
        try: script.unload()
        except Exception: pass
        try: sess.detach()
        except Exception: pass


if __name__ == '__main__':
    main()
