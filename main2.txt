#!/usr/bin/env python3
"""
Frida script to find processes with 'detect' in name and monitor network traffic.
Prints network events and attempts to decrypt common encryption schemes.
"""

import frida
import sys
import json
import time
import re
import base64
import hashlib
from datetime import datetime

try:
    from Crypto.Cipher import AES, DES, ARC4
    from Crypto.Util.Padding import unpad
    HAVE_PYCRYPTO = True
except ImportError:
    HAVE_PYCRYPTO = False
    print("[!] pycryptodome not installed - AES/DES/RC4 decryption disabled")
    print("    Install with: pip install pycryptodome")


# ---------------------------------------------------------------------------
# Frida agent - hooks network APIs across platforms
# ---------------------------------------------------------------------------
JS_AGENT = r"""
'use strict';

const seen = new Set();

function ts() {
    return new Date().toISOString();
}

function emit(payload) {
    try {
        send(payload);
    } catch (e) {
        console.log('[!] send failed: ' + e);
    }
}

function bytesToHex(bytes) {
    let s = '';
    for (let i = 0; i < bytes.length; i++) {
        s += ('0' + (bytes[i] & 0xff).toString(16)).slice(-2);
    }
    return s;
}

function readMemory(ptr, len) {
    try {
        if (!ptr || ptr.isNull() || len <= 0 || len > 0x100000) return null;
        return new Uint8Array(ptr.readByteArray(len));
    } catch (e) {
        return null;
    }
}

function u8ToStr(u8) {
    try {
        let s = '';
        for (let i = 0; i < u8.length; i++) {
            const c = u8[i];
            if (c === 0) break;
            s += String.fromCharCode(c);
        }
        return s;
    } catch (e) { return ''; }
}

function emitEvent(kind, data) {
    emit({
        type: 'network_event',
        kind: kind,
        timestamp: ts(),
        data: data
    });
}

// -----------------------------------------------------------------
// Hook libc send/recv/sendto/recvfrom
// -----------------------------------------------------------------
function hookLibc() {
    let libcName = null;
    if (Process.platform === 'windows') libcName = 'ws2_32.dll';
    else if (Process.platform === 'darwin') libcName = 'libsystem_c.dylib';
    else libcName = 'libc.so.6';

    let libc = null;
    try { libc = Process.getModuleByName(libcName); } catch (e) {}
    if (!libc) {
        try { libc = Process.getModuleByName('libc.so'); } catch (e) {}
    }
    if (!libc) {
        console.log('[!] could not find libc-like module');
        return;
    }

    const funcs = Process.platform === 'windows'
        ? [['send', 'int'], ['recv', 'int'], ['sendto', 'int'], ['recvfrom', 'int'], ['WSASend', 'int'], ['WSARecv', 'int']]
        : [['send', 'ssize_t'], ['recv', 'ssize_t'], ['sendto', 'ssize_t'], ['recvfrom', 'ssize_t'],
           ['write', 'ssize_t'], ['read', 'ssize_t']];

    for (const [name, ret] of funcs) {
        let addr = null;
        try { addr = Module.findExportByName(libcName, name); } catch (e) {}
        if (!addr) {
            try { addr = Module.findExportByName(null, name); } catch (e) {}
        }
        if (!addr) continue;

        try {
            if (name === 'send' || name === 'sendto' || name === 'write' || name === 'WSASend') {
                Interceptor.attach(addr, {
                    onEnter(args) {
                        const fd = args[0].toInt32();
                        const buf = args[1];
                        let len = args[2].toInt32();
                        // WSASend: buf is WSABUF* structure
                        if (name === 'WSASend') {
                            try {
                                const wsabuf = args[1];
                                const realLen = wsabuf.readU32();
                                const realBuf = wsabuf.add(Process.pointerSize).readPointer();
                                const data = readMemory(realBuf, realLen);
                                if (data) {
                                    emitEvent('outgoing', {
                                        api: name,
                                        fd: fd,
                                        length: realLen,
                                        hex: bytesToHex(data),
                                        ascii: u8ToStr(data)
                                    });
                                }
                            } catch (e) {}
                            return;
                        }
                        if (len <= 0 || len > 0x100000) return;
                        const data = readMemory(buf, len);
                        if (!data) return;
                        emitEvent('outgoing', {
                            api: name,
                            fd: fd,
                            length: len,
                            hex: bytesToHex(data),
                            ascii: u8ToStr(data)
                        });
                    }
                });
            } else {
                // recv/read/recvfrom/WSARecv
                Interceptor.attach(addr, {
                    onEnter(args) {
                        this.fd = args[0].toInt32();
                        this.buf = args[1];
                        this.len = args[2].toInt32();
                        this.api = name;
                    },
                    onLeave(retval) {
                        const n = retval.toInt32();
                        if (n <= 0 || n > 0x100000) return;
                        let data = null;
                        if (this.api === 'WSARecv') {
                            try {
                                const wsabuf = this.buf;
                                const realLen = wsabuf.readU32();
                                const realBuf = wsabuf.add(Process.pointerSize).readPointer();
                                data = readMemory(realBuf, realLen);
                            } catch (e) {}
                        } else {
                            data = readMemory(this.buf, n);
                        }
                        if (!data) return;
                        emitEvent('incoming', {
                            api: this.api,
                            fd: this.fd,
                            length: n,
                            hex: bytesToHex(data),
                            ascii: u8ToStr(data)
                        });
                    }
                });
            }
        } catch (e) {
            console.log('[!] failed to hook ' + name + ': ' + e);
        }
    }
}

// -----------------------------------------------------------------
// Hook OpenSSL SSL_write / SSL_read (search all modules)
// -----------------------------------------------------------------
function hookOpenSSL() {
    const symbols = ['SSL_write', 'SSL_read', 'SSL_write_ex', 'SSL_read_ex'];
    const modules = Process.enumerateModules();
    const targets = [];

    for (const m of modules) {
        const n = m.name.toLowerCase();
        if (n.includes('ssl') || n.includes('crypto') || n.includes('tls') || n.includes('curl') || n.includes('nspr') || n.includes('nss')) {
            for (const sym of symbols) {
                try {
                    const a = Module.findExportByName(m.name, sym);
                    if (a) targets.push({ sym: sym, addr: a, module: m.name });
                } catch (e) {}
            }
        }
    }

    for (const t of targets) {
        const key = t.module + '!' + t.sym;
        if (seen.has(key)) continue;
        seen.add(key);

        try {
            if (t.sym.endsWith('_ex')) {
                Interceptor.attach(t.addr, {
                    onEnter(args) {
                        this.ssl = args[0];
                        this.buf = args[1];
                        this.lenPtr = args[2];
                    },
                    onLeave(retval) {
                        try {
                            if (retval.toInt32() !== 1) return;
                            const n = this.lenPtr.readULong();
                            if (n <= 0 || n > 0x100000) return;
                            const data = readMemory(this.buf, n);
                            if (!data) return;
                            emitEvent(t.sym.indexOf('write') !== -1 ? 'outgoing_tls' : 'incoming_tls', {
                                api: key,
                                length: n,
                                hex: bytesToHex(data),
                                ascii: u8ToStr(data)
                            });
                        } catch (e) {}
                    }
                });
            } else {
                Interceptor.attach(t.addr, {
                    onEnter(args) {
                        this.ssl = args[0];
                        this.buf = args[1];
                        this.len = args[2].toInt32();
                        this.isWrite = t.sym.indexOf('write') !== -1;
                    },
                    onLeave(retval) {
                        const n = retval.toInt32();
                        if (n <= 0 || n > 0x100000) return;
                        const data = readMemory(this.buf, n);
                        if (!data) return;
                        emitEvent(this.isWrite ? 'outgoing_tls' : 'incoming_tls', {
                            api: key,
                            length: n,
                            hex: bytesToHex(data),
                            ascii: u8ToStr(data)
                        });
                    }
                });
            }
            console.log('[*] hooked ' + key);
        } catch (e) {
            console.log('[!] hook failed for ' + key + ': ' + e);
        }
    }
}

// -----------------------------------------------------------------
// Hook Java (Android) OkHttp / HttpURLConnection / SSLSocket
// -----------------------------------------------------------------
function hookJava() {
    if (!Java.available) return;
    Java.perform(function () {
        try {
            const Buffer = Java.use('okio.Buffer');
            const Okio = Java.use('okio.Okio');
            const RequestBody = Java.use('okhttp3.RequestBody');
            const ResponseBody = Java.use('okhttp3.ResponseBody');
            const RealCall = Java.use('okhttp3.internal.connection.RealCall');

            // Hook response body bytes
            ResponseBody.bytes.implementation = function () {
                const b = this.bytes();
                try {
                    emitEvent('http_response_java', {
                        api: 'okhttp3.ResponseBody.bytes',
                        length: b.length,
                        ascii: Java.array_to_string ? Java.array_to_string(b) : null,
                        base64: send({ __b64: true, v: b })
                    });
                } catch (e) {}
                return b;
            };
        } catch (e) {
            // OkHttp not present
        }

        try {
            const SSLSocket = Java.use('javax.net.ssl.SSLSocket');
            const SSLSocketImpl = Java.use('sun.security.ssl.SSLSocketImpl');
            // Fallback via Conscrypt
        } catch (e) {}

        // Generic Java socket streams
        try {
            const FileOutputStream = Java.use('java.io.FileOutputStream');
            const OutputStream = Java.use('java.io.OutputStream');
            const InputStream = Java.use('java.io.InputStream');

            OutputStream.write.overload('[B').implementation = function (b) {
                try {
                    const s = Java.array_to_string ? Java.array_to_string(b) : '';
                    emitEvent('java_out', {
                        api: this.getClass().getName() + '.write',
                        length: b.length,
                        ascii: s
                    });
                } catch (e) {}
                return this.write(b);
            };
        } catch (e) {}
    });
}

// -----------------------------------------------------------------
// Windows: WinHTTP / WinINet hooks
// -----------------------------------------------------------------
function hookWindowsHttp() {
    if (Process.platform !== 'windows') return;
    const mods = ['winhttp.dll', 'wininet.dll'];
    for (const m of mods) {
        try {
            const mod = Process.getModuleByName(m);
        } catch (e) { continue; }
        const syms = m === 'winhttp.dll'
            ? ['WinHttpWriteData', 'WinHttpReadData', 'WinHttpSendRequest', 'WinHttpReceiveResponse']
            : ['InternetWriteFile', 'InternetReadFile', 'HttpSendRequestA', 'HttpSendRequestW'];

        for (const sym of syms) {
            const addr = Module.findExportByName(m, sym);
            if (!addr) continue;
            const isWrite = sym.toLowerCase().includes('write') || sym.toLowerCase().includes('send');
            try {
                if (isWrite) {
                    Interceptor.attach(addr, {
                        onEnter(args) {
                            let buf, len;
                            if (sym === 'WinHttpWriteData') { buf = args[1]; len = args[2].toInt32(); }
                            else if (sym === 'WinHttpSendRequest') { buf = args[1]; len = args[2].toInt32(); }
                            else if (sym === 'InternetWriteFile') { buf = args[1]; len = args[2].toInt32(); }
                            else return;
                            if (len <= 0) return;
                            const data = readMemory(buf, len);
                            if (!data) return;
                            emitEvent('outgoing_winhttp', {
                                api: m + '!' + sym,
                                length: len,
                                hex: bytesToHex(data),
                                ascii: u8ToStr(data)
                            });
                        }
                    });
                } else {
                    Interceptor.attach(addr, {
                        onEnter(args) {
                            this.buf = args[1];
                            this.lenPtr = args[2];
                            this.api = sym;
                        },
                        onLeave(retval) {
                            try {
                                let n = this.lenPtr.readU32();
                                if (n <= 0 || n > 0x100000) return;
                                const data = readMemory(this.buf, n);
                                if (!data) return;
                                emitEvent('incoming_winhttp', {
                                    api: m + '!' + this.api,
                                    length: n,
                                    hex: bytesToHex(data),
                                    ascii: u8ToStr(data)
                                });
                            } catch (e) {}
                        }
                    });
                }
                console.log('[*] hooked ' + m + '!' + sym);
            } catch (e) {}
        }
    }
}

// -----------------------------------------------------------------
// Entry
// -----------------------------------------------------------------
setImmediate(function () {
    console.log('[*] detect_monitor agent loaded into ' + Process.platform + ' pid ' + Process.id);
    hookLibc();
    hookOpenSSL();
    hookWindowsHttp();
    hookJava();
    console.log('[*] hooks installed');
});
"""


# ---------------------------------------------------------------------------
# Decryption helpers (Python side)
# ---------------------------------------------------------------------------
HEX_RE = re.compile(r'^[0-9a-fA-F]+$')


def looks_like_json(s: str) -> bool:
    s = s.strip()
    if not s or s[0] not in '{[':
        return False
    try:
        json.loads(s)
        return True
    except Exception:
        return False


def try_decrypt_blob(data: bytes):
    """
    Attempt common decryption schemes against a ciphertext blob.
    Returns a list of (scheme, plaintext_bytes).
    """
    results = []

    if not HAVE_PYCRYPTO:
        return results

    # Try AES-ECB with common keys
    common_keys = [
        b'0123456789abcdef', b'0123456789ABCDEF',
        b'0000000000000000', b'\x00' * 16,
        b'1234567890123456', b'passwordpassword',
        b'changeit_change1',
    ]
    iv_zero = b'\x00' * 16

    for k in common_keys:
        # AES ECB
        try:
            c = AES.new(k, AES.MODE_ECB)
            pt = c.decrypt(data)
            try:
                pt_u = unpad(pt, 16)
            except Exception:
                pt_u = pt
            if is_printable(pt_u):
                results.append(('AES-ECB key=' + k.hex(), pt_u))
        except Exception:
            pass
        # AES CBC zero IV
        try:
            c = AES.new(k, AES.MODE_CBC, iv_zero)
            pt = c.decrypt(data)
            try:
                pt_u = unpad(pt, 16)
            except Exception:
                pt_u = pt
            if is_printable(pt_u):
                results.append(('AES-CBC zeroIV key=' + k.hex(), pt_u))
        except Exception:
            pass

    # RC4 with common keys
    for k in [b'key', b'secret', b'password', b'0123456789abcdef']:
        try:
            c = ARC4.new(k)
            pt = c.decrypt(data)
            if is_printable(pt):
                results.append(('RC4 key=' + k.hex(), pt))
        except Exception:
            pass

    # XOR single-byte brute
    for key in range(1, 256):
        pt = bytes(b ^ key for b in data)
        if looks_like_json(pt.decode('utf-8', errors='ignore')):
            results.append(('XOR key=0x%02x' % key, pt))
            break

    # Base64 layer
    try:
        s = data.decode('ascii')
        if re.fullmatch(r'[A-Za-z0-9+/=\s]+', s) and len(s) % 4 == 0:
            dec = base64.b64decode(s)
            if is_printable(dec):
                results.append(('Base64', dec))
    except Exception:
        pass

    return results


def is_printable(b: bytes, threshold: float = 0.85) -> bool:
    if not b:
        return False
    printable = sum(1 for c in b if 32 <= c < 127 or c in (9, 10, 13))
    return (printable / len(b)) >= threshold


# ---------------------------------------------------------------------------
# Process discovery
# ---------------------------------------------------------------------------
def find_detect_processes():
    matches = []
    for proc in frida.enumerate_processes():
        if 'detect' in proc.name.lower():
            matches.append(proc)
    return matches


# ---------------------------------------------------------------------------
# Event handling
# ---------------------------------------------------------------------------
def on_message(message, data):
    if message.get('type') == 'send':
        payload = message.get('payload')

        # Handle Java byte array b64 payloads
        if isinstance(payload, dict) and payload.get('__b64'):
            return

        if not isinstance(payload, dict):
            print('[raw] ' + str(payload))
            return

        if payload.get('type') == 'network_event':
            handle_network_event(payload)
        else:
            print(json.dumps(payload, indent=2))
    elif message.get('type') == 'error':
        print('[frida error] ' + message.get('description', ''))
        print(message.get('stack', ''))


def handle_network_event(evt):
    kind = evt.get('kind')
    data = evt.get('data', {})
    t = evt.get('timestamp', '')

    # Build header
    api = data.get('api', '?')
    fd = data.get('fd', '?')
    length = data.get('length', '?')
    ascii_preview = (data.get('ascii', '') or '')[:200]
    hex_str = data.get('hex', '')

    print('\n' + '=' * 78)
    print(f'[{t}] {kind.upper():<18} {api}  fd={fd}  len={length}')
    print('-' * 78)

    # Show ASCII preview
    if ascii_preview:
        print('ASCII: ' + repr(ascii_preview))

    # Try JSON directly from ascii
    if looks_like_json(data.get('ascii', '')):
        try:
            j = json.loads(data.get('ascii', ''))
            print('JSON (plaintext):')
            print(json.dumps(j, indent=2)[:4000])
        except Exception:
            pass

    # Decryption attempts
    if hex_str and HEX_RE.match(hex_str):
        try:
            raw = bytes.fromhex(hex_str)
        except Exception:
            raw = b''

        if raw:
            # If it's already printable, no need to decrypt
            if not is_printable(raw):
                print('[*] ciphertext (not printable) - attempting decryption...')
                results = try_decrypt_blob(raw)
                if results:
                    for scheme, pt in results:
                        print(f'[+] DECRYPTED via {scheme}:')
                        try:
                            txt = pt.decode('utf-8', errors='replace')
                        except Exception:
                            txt = repr(pt)
                        # Truncate
                        print('    ' + txt[:2000].replace('\n', '\n    '))
                        # Highlight JSON
                        if looks_like_json(txt):
                            try:
                                j = json.loads(txt)
                                print('    JSON:')
                                print('    ' + json.dumps(j, indent=2)[:2000].replace('\n', '\n    '))
                            except Exception:
                                pass
                else:
                    print('[*] no decryption succeeded with common schemes')
            else:
                # Try to extract embedded JSON
                text = raw.decode('utf-8', errors='replace')
                if looks_like_json(text):
                    try:
                        print('JSON:')
                        print(json.dumps(json.loads(text), indent=2)[:4000])
                    except Exception:
                        pass


# ---------------------------------------------------------------------------
# Main
# ---------------------------------------------------------------------------
def attach_and_monitor(proc, session_holder):
    print(f'\n[+] Attaching to {proc.name} (pid={proc.pid})')
    try:
        session = frida.attach(proc.pid)
    except Exception as e:
        print(f'[!] attach failed: {e}')
        return None

    script = session.create_script(JS_AGENT)
    script.on('message', on_message)
    script.load()

    session_holder.append((proc, session, script))
    print(f'[+] Instrumented pid={proc.pid}')
    return session


def main():
    print('[*] Searching for processes containing "detect"...')
    matches = find_detect_processes()

    if not matches:
        print('[!] No processes found with "detect" in name.')
        print('    Will keep watching and attach to new ones as they appear.')
    else:
        for p in matches:
            print(f'    - {p.name} (pid={p.pid})')

    sessions = []
    for p in matches:
        attach_and_monitor(p, sessions)

    if not sessions:
        # Polling loop for new processes
        print('[*] Polling for new "detect" processes every 2s (Ctrl+C to quit)...')
        known = set()
        try:
            while True:
                for p in find_detect_processes():
                    if p.pid not in known:
                        known.add(p.pid)
                        attach_and_monitor(p, sessions)
                time.sleep(2)
        except KeyboardInterrupt:
            print('\n[*] Stopping...')
    else:
        print('\n[*] Monitoring. Press Ctrl+C to exit.')
        try:
            sys.stdin.read()
        except KeyboardInterrupt:
            pass

    for proc, sess, script in sessions:
        try:
            script.unload()
        except Exception:
            pass
        try:
            sess.detach()
        except Exception:
            pass


if __name__ == '__main__':
    main()
