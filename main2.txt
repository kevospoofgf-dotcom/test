#!/usr/bin/env python3
"""
Capture decrypted TLS traffic from detect-fm-<PIN>.exe.

Hooks sspicli/secur32!EncryptMessage and DecryptMessage (system exports only
— app module is never touched), parses the PSecBufferDesc correctly, and
prints the plaintext HTTP/JSON before encryption and after decryption.
"""

import frida
import sys
import time
import json
import re
import psutil

try:
    from Crypto.Cipher import AES
    HAVE_PYCRYPTO = True
except ImportError:
    HAVE_PYCRYPTO = False


JS = r"""
'use strict';

function emit(p) { try { send(p); } catch (e) {} }

function ts() { return new Date().toISOString(); }

function bytesToHex(b) {
    let s = '';
    for (let i = 0; i < b.length; i++)
        s += ('0' + (b[i] & 0xff).toString(16)).slice(-2);
    return s;
}

function readMem(ptr, len) {
    try {
        if (!ptr || ptr.isNull() || len <= 0 || len > 0x100000) return null;
        return new Uint8Array(ptr.readByteArray(len));
    } catch (e) { return null; }
}

function u8ToStr(u8) {
    try {
        let s = '';
        for (let i = 0; i < u8.length; i++) {
            const c = u8[i];
            if (c === 0) break;
            s += String.fromCharCode(c);
        }
        return s;
    } catch (e) { return ''; }
}

function emitEvent(kind, data) {
    emit({ type: 'network_event', kind: kind, timestamp: ts(), data: data });
}

function getGlobalExport(sym) {
    try {
        if (typeof Module.findGlobalExportByName === 'function')
            return Module.findGlobalExportByName(sym);
    } catch (e) {}
    try { return Module.findExportByName(null, sym); } catch (e) {}
    return null;
}

function getExport(mod, sym) {
    try { return Module.getExportByName(mod, sym); } catch (e) {}
    try { return Module.findExportByName(mod, sym); } catch (e) {}
    return null;
}

// Correct SecBufferDesc layout:
//   +0  ULONG ulVersion    (usually 0)
//   +4  ULONG cBuffers     (count)
//   +8  PSecBuffer pBuffers (array of {cbBuffer, BufferType, pvBuffer})
// SecBuffer on x64 is 16 bytes with padding.
function parseSecBufferDesc(pDesc, api, direction) {
    try {
        if (!pDesc || pDesc.isNull()) return;
        const ulVersion = pDesc.readU32();
        const cBuffers  = pDesc.add(4).readU32();
        if (cBuffers < 1 || cBuffers > 16) return;
        const bufArray = pDesc.add(8).readPointer();
        if (!bufArray || bufArray.isNull()) return;

        const stride = 16;
        for (let i = 0; i < cBuffers; i++) {
            const entry = bufArray.add(i * stride);
            let cbBuffer = 0, bufType = 0, pvBuffer = null;
            try {
                cbBuffer = entry.readU32();
                bufType  = entry.add(4).readU32();
                pvBuffer = entry.add(8).readPointer();
            } catch (e) { continue; }

            // 0x01 = SECBUFFER_DATA, 0x07 = SECBUFFER_STREAM
            if (bufType !== 0x01 && bufType !== 0x07) continue;
            if (cbBuffer === 0 || cbBuffer > 0x100000) continue;

            const data = readMem(pvBuffer, cbBuffer);
            if (!data) continue;

            emitEvent(
                direction === 'out' ? 'outgoing_schannel_plain'
                                    : 'incoming_schannel_plain',
                {
                    api: api,
                    length: cbBuffer,
                    hex: bytesToHex(data),
                    ascii: u8ToStr(data)
                }
            );
        }
    } catch (e) {}
}

let encCount = 0, decCount = 0;

function hookSspi() {
    // EncryptMessage(PCtxtHandle, ULONG, PSecBufferDesc, ULONG)
    //   → args[2] is the PSecBufferDesc
    const encAddr = getGlobalExport('EncryptMessage')
                 || getExport('secur32.dll', 'EncryptMessage')
                 || getExport('sspicli.dll', 'EncryptMessage');
    if (encAddr) {
        try {
            Interceptor.attach(encAddr, {
                onEnter(args) {
                    encCount++;
                    // On Encrypt, the DATA buffer holds plaintext here.
                    parseSecBufferDesc(args[2], 'EncryptMessage', 'out');
                }
            });
            console.log('[*] hooked EncryptMessage @ ' + encAddr);
        } catch (e) { console.log('[!] hook EncryptMessage: ' + e); }
    } else {
        console.log('[!] EncryptMessage not resolved');
    }

    // DecryptMessage(PCtxtHandle, PSecBufferDesc, ULONG, PULONG)
    //   → args[1] is the PSecBufferDesc
    const decAddr = getGlobalExport('DecryptMessage')
                 || getExport('secur32.dll', 'DecryptMessage')
                 || getExport('sspicli.dll', 'DecryptMessage');
    if (decAddr) {
        try {
            Interceptor.attach(decAddr, {
                onEnter(args) {
                    this.pDesc = args[1];
                },
                onLeave(retval) {
                    // SEC_E_OK == 0; the plaintext is now in the DATA buffer.
                    if (retval.toInt32() !== 0) return;
                    decCount++;
                    parseSecBufferDesc(this.pDesc, 'DecryptMessage', 'in');
                }
            });
            console.log('[*] hooked DecryptMessage @ ' + decAddr);
        } catch (e) { console.log('[!] hook DecryptMessage: ' + e); }
    } else {
        console.log('[!] DecryptMessage not resolved');
    }
}

function waitForModule(name, cb, tries) {
    tries = tries || 0;
    try {
        if (Process.getModuleByName(name)) { cb(); return; }
    } catch (e) {}
    if (tries > 50) { cb(); return; }
    setTimeout(function () { waitForModule(name, cb, tries + 1); }, 100);
}

setImmediate(function () {
    console.log('[*] capture agent v6 loaded (pid=' + Process.id + ')');
    waitForModule('secur32.dll', function () {
        hookSspi();
        console.log('[*] hooks installed');
    });
});
"""


def looks_like_json(s):
    s = (s or '').strip()
    if not s or s[0] not in '{[':
        return False
    try:
        json.loads(s)
        return True
    except Exception:
        return False


def on_message(message, data):
    if message.get('type') != 'send':
        if message.get('type') == 'error':
            print('[frida error] ' + message.get('description', ''))
        return
    p = message.get('payload')
    if not isinstance(p, dict):
        return
    if p.get('type') != 'network_event':
        return

    kind = p.get('kind', '')
    data = p.get('data', {})
    t = p.get('timestamp', '')
    api = data.get('api', '?')
    length = data.get('length', '?')
    ascii_str = data.get('ascii', '') or ''

    # Only print plaintext that looks like HTTP or JSON
    printable_run = 0
    max_run = 0
    for ch in ascii_str[:256]:
        c = ord(ch)
        if 32 <= c < 127 or c in (9, 10, 13):
            printable_run += 1
            if printable_run > max_run: max_run = printable_run
        else:
            printable_run = 0
    if max_run < 16:
        return

    print('\n' + '=' * 78)
    print(f'[{t}] {kind.upper():<26} {api}  len={length}')
    print('-' * 78)
    print('ASCII: ' + repr(ascii_str[:600]))
    if looks_like_json(ascii_str):
        try:
            print('JSON:')
            print(json.dumps(json.loads(ascii_str), indent=2)[:4000])
        except Exception:
            pass


def find_detect_processes():
    out = []
    for proc in psutil.process_iter(['pid', 'name']):
        try:
            name = (proc.info.get('name') or '').lower()
            if 'detect-fm' in name:
                out.append(proc.info)
        except (psutil.NoSuchProcess, psutil.AccessDenied):
            pass
    return out


def attach_and_run(pid, name):
    print(f'\n[+] Attaching to {name} (pid={pid})')
    try:
        session = frida.attach(pid)
    except Exception as e:
        print(f'[!] attach failed: {e}')
        return None
    script = session.create_script(JS)
    script.on('message', on_message)
    try:
        script.load()
    except Exception as e:
        print(f'[!] script.load failed: {e}')
        try: session.detach()
        except Exception: pass
        return None
    print(f'[+] Instrumented pid={pid}')
    return session, script


def main():
    device = frida.get_local_device()
    print(f'[*] Using device: {device}')

    existing = find_detect_processes()
    sessions = []
    for e in existing:
        r = attach_and_run(e['pid'], e['name'])
        if r:
            sessions.append(r)

    if not sessions:
        print('[*] Polling every 2s (Ctrl+C to quit)...')
        known = set()
        try:
            while True:
                for e in find_detect_processes():
                    if e['pid'] in known:
                        continue
                    r = attach_and_run(e['pid'], e['name'])
                    if r:
                        known.add(e['pid'])
                        sessions.append(r)
                time.sleep(2)
        except KeyboardInterrupt:
            pass
    else:
        print('\n[*] Monitoring. Press Ctrl+C to exit.')
        try:
            while True:
                time.sleep(1)
        except KeyboardInterrupt:
            pass

    for sess, script in sessions:
        try: script.unload()
        except Exception: pass
        try: sess.detach()
        except Exception: pass


if __name__ == '__main__':
    main()
