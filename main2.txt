#!/usr/bin/env python3
"""
Capture decrypted TLS traffic from detect-fm-<PIN>.exe.

Hooks sspicli/secur32!EncryptMessage and DecryptMessage (system exports only
— app module is never touched), parses the PSecBufferDesc correctly, and
logs plaintext HTTP/JSON before encryption and after decryption.

Writes to %TEMP%\\debug.log (live) and %TEMP%\\debug.jsonl (structured).
"""

import frida
import sys
import time
import json
import re
import os
import tempfile
import psutil

try:
    from Crypto.Cipher import AES
    HAVE_PYCRYPTO = True
except ImportError:
    HAVE_PYCRYPTO = False


# ---------------------------------------------------------------------------
# Log file paths — %TEMP%\debug.log on Windows, /tmp/debug.log elsewhere.
# ---------------------------------------------------------------------------
_LOG_DIR = tempfile.gettempdir()
_LOG_TXT = os.path.join(_LOG_DIR, 'debug.log')
_LOG_JSONL = os.path.join(_LOG_DIR, 'debug.jsonl')

# Truncate at start of each run so the file is clean for the next upload.
try:
    with open(_LOG_TXT, 'w', encoding='utf-8') as f:
        f.write(f'# detect-fm capture — started {time.strftime("%Y-%m-%d %H:%M:%S")}\n')
    with open(_LOG_JSONL, 'w', encoding='utf-8') as f:
        f.write('')
except Exception as e:
    print(f'[!] could not open log files: {e}')


_LOG_TXT_F = open(_LOG_TXT, 'a', encoding='utf-8', errors='replace')
_LOG_JSONL_F = open(_LOG_JSONL, 'a', encoding='utf-8', errors='replace')


def log_write_txt(s):
    """Append a line to debug.log with a timestamp, flush immediately."""
    try:
        ts = time.strftime('%H:%M:%S')
        _LOG_TXT_F.write(f'[{ts}] {s}\n')
        _LOG_TXT_F.flush()
    except Exception:
        pass


def log_write_jsonl(obj):
    """Append a JSON object as one line to debug.jsonl."""
    try:
        _LOG_JSONL_F.write(json.dumps(obj, ensure_ascii=False) + '\n')
        _LOG_JSONL_F.flush()
    except Exception:
        pass


JS = r"""
'use strict';

function emit(p) { try { send(p); } catch (e) {} }
function ts() { return new Date().toISOString(); }

function bytesToHex(b) {
    let s = '';
    for (let i = 0; i < b.length; i++)
        s += ('0' + (b[i] & 0xff).toString(16)).slice(-2);
    return s;
}

function readMem(ptr, len) {
    try {
        if (!ptr || ptr.isNull() || len <= 0 || len > 0x100000) return null;
        return new Uint8Array(ptr.readByteArray(len));
    } catch (e) { return null; }
}

function u8ToStr(u8) {
    try {
        let s = '';
        for (let i = 0; i < u8.length; i++) {
            const c = u8[i];
            if (c === 0) break;
            s += String.fromCharCode(c);
        }
        return s;
    } catch (e) { return ''; }
}

function emitEvent(kind, data) {
    emit({ type: 'network_event', kind: kind, timestamp: ts(), data: data });
}

function getGlobalExport(sym) {
    try {
        if (typeof Module.findGlobalExportByName === 'function')
            return Module.findGlobalExportByName(sym);
    } catch (e) {}
    try { return Module.findExportByName(null, sym); } catch (e) {}
    return null;
}

function getExport(mod, sym) {
    try { return Module.getExportByName(mod, sym); } catch (e) {}
    try { return Module.findExportByName(mod, sym); } catch (e) {}
    return null;
}

function parseSecBufferDesc(pDesc, api, direction) {
    try {
        if (!pDesc || pDesc.isNull()) return;
        const ulVersion = pDesc.readU32();
        const cBuffers  = pDesc.add(4).readU32();
        if (cBuffers < 1 || cBuffers > 16) return;
        const bufArray = pDesc.add(8).readPointer();
        if (!bufArray || bufArray.isNull()) return;

        const stride = 16;
        for (let i = 0; i < cBuffers; i++) {
            const entry = bufArray.add(i * stride);
            let cbBuffer = 0, bufType = 0, pvBuffer = null;
            try {
                cbBuffer = entry.readU32();
                bufType  = entry.add(4).readU32();
                pvBuffer = entry.add(8).readPointer();
            } catch (e) { continue; }

            if (bufType !== 0x01 && bufType !== 0x07) continue;
            if (cbBuffer === 0 || cbBuffer > 0x100000) continue;

            const data = readMem(pvBuffer, cbBuffer);
            if (!data) continue;

            emitEvent(
                direction === 'out' ? 'outgoing_schannel_plain'
                                    : 'incoming_schannel_plain',
                {
                    api: api,
                    length: cbBuffer,
                    hex: bytesToHex(data),
                    ascii: u8ToStr(data)
                }
            );
        }
    } catch (e) {}
}

function hookSspi() {
    const encAddr = getGlobalExport('EncryptMessage')
                 || getExport('secur32.dll', 'EncryptMessage')
                 || getExport('sspicli.dll', 'EncryptMessage');
    if (encAddr) {
        Interceptor.attach(encAddr, {
            onEnter(args) {
                parseSecBufferDesc(args[2], 'EncryptMessage', 'out');
            }
        });
        console.log('[*] hooked EncryptMessage @ ' + encAddr);
    } else {
        console.log('[!] EncryptMessage not resolved');
    }

    const decAddr = getGlobalExport('DecryptMessage')
                 || getExport('secur32.dll', 'DecryptMessage')
                 || getExport('sspicli.dll', 'DecryptMessage');
    if (decAddr) {
        Interceptor.attach(decAddr, {
            onEnter(args) {
                this.pDesc = args[1];
            },
            onLeave(retval) {
                if (retval.toInt32() !== 0) return;
                parseSecBufferDesc(this.pDesc, 'DecryptMessage', 'in');
            }
        });
        console.log('[*] hooked DecryptMessage @ ' + decAddr);
    } else {
        console.log('[!] DecryptMessage not resolved');
    }
}

function waitForModule(name, cb, tries) {
    tries = tries || 0;
    try {
        if (Process.getModuleByName(name)) { cb(); return; }
    } catch (e) {}
    if (tries > 50) { cb(); return; }
    setTimeout(function () { waitForModule(name, cb, tries + 1); }, 100);
}

setImmediate(function () {
    console.log('[*] capture agent v7 loaded (pid=' + Process.id + ')');
    waitForModule('secur32.dll', function () {
        hookSspi();
        console.log('[*] hooks installed');
    });
});
"""


def looks_like_json(s):
    s = (s or '').strip()
    if not s or s[0] not in '{[':
        return False
    try:
        json.loads(s)
        return True
    except Exception:
        return False


def on_message(message, data):
    if message.get('type') != 'send':
        if message.get('type') == 'error':
            msg = '[frida error] ' + message.get('description', '')
            print(msg)
            log_write_txt(msg)
        return

    p = message.get('payload')
    if not isinstance(p, dict):
        return
    if p.get('type') != 'network_event':
        return

    kind = p.get('kind', '')
    payload_data = p.get('data', {})
    t = p.get('timestamp', '')
    api = payload_data.get('api', '?')
    length = payload_data.get('length', '?')
    ascii_str = payload_data.get('ascii', '') or ''
    hex_str = payload_data.get('hex', '')

    # Filter: only keep events with a readable ASCII run
    printable_run = 0
    max_run = 0
    for ch in ascii_str[:512]:
        c = ord(ch)
        if 32 <= c < 127 or c in (9, 10, 13):
            printable_run += 1
            if printable_run > max_run: max_run = printable_run
        else:
            printable_run = 0

    is_interesting = max_run >= 8

    # Always write to JSONL (even uninteresting — useful for reviewing raw bytes later)
    log_write_jsonl({
        'ts': t,
        'kind': kind,
        'api': api,
        'length': length,
        'ascii': ascii_str,
        'hex': hex_str,
    })

    if not is_interesting:
        return

    # Console + text log for readable content
    header = f'\n{"=" * 78}\n[{t}] {kind.upper():<26} {api}  len={length}\n{"-" * 78}'
    print(header)
    print('ASCII: ' + repr(ascii_str[:600]))
    log_write_txt(header)
    log_write_txt('ASCII: ' + repr(ascii_str[:600]))

    if looks_like_json(ascii_str):
        try:
            pretty = json.dumps(json.loads(ascii_str), indent=2)[:4000]
            print('JSON:\n' + pretty)
            log_write_txt('JSON:\n' + pretty)
        except Exception:
            pass


def find_detect_processes():
    out = []
    for proc in psutil.process_iter(['pid', 'name']):
        try:
            name = (proc.info.get('name') or '').lower()
            if 'detect-fm' in name:
                out.append(proc.info)
        except (psutil.NoSuchProcess, psutil.AccessDenied):
            pass
    return out


def attach_and_run(pid, name):
    msg = f'\n[+] Attaching to {name} (pid={pid})'
    print(msg)
    log_write_txt(msg)
    try:
        session = frida.attach(pid)
    except Exception as e:
        m = f'[!] attach failed: {e}'
        print(m); log_write_txt(m)
        return None
    script = session.create_script(JS)
    script.on('message', on_message)
    try:
        script.load()
    except Exception as e:
        m = f'[!] script.load failed: {e}'
        print(m); log_write_txt(m)
        try: session.detach()
        except Exception: pass
        return None
    m = f'[+] Instrumented pid={pid}'
    print(m); log_write_txt(m)
    return session, script


def main():
    print(f'[*] Writing logs to:')
    print(f'    text   -> {_LOG_TXT}')
    print(f'    jsonl  -> {_LOG_JSONL}')
    log_write_txt(f'[*] text log:  {_LOG_TXT}')
    log_write_txt(f'[*] jsonl log: {_LOG_JSONL}')

    device = frida.get_local_device()
    print(f'[*] Using device: {device}')

    existing = find_detect_processes()
    sessions = []
    for e in existing:
        r = attach_and_run(e['pid'], e['name'])
        if r:
            sessions.append(r)

    if not sessions:
        print('[*] Polling every 2s (Ctrl+C to quit)...')
        known = set()
        try:
            while True:
                for e in find_detect_processes():
                    if e['pid'] in known:
                        continue
                    r = attach_and_run(e['pid'], e['name'])
                    if r:
                        known.add(e['pid'])
                        sessions.append(r)
                time.sleep(2)
        except KeyboardInterrupt:
            pass
    else:
        print('\n[*] Monitoring. Press Ctrl+C to exit.')
        try:
            while True:
                time.sleep(1)
        except KeyboardInterrupt:
            pass

    for sess, script in sessions:
        try: script.unload()
        except Exception: pass
        try: sess.detach()
        except Exception: pass

    log_write_txt('[*] session ended')
    try:
        _LOG_TXT_F.close()
        _LOG_JSONL_F.close()
    except Exception:
        pass


if __name__ == '__main__':
    main()
