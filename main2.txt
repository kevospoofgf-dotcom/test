#!/usr/bin/env python3
"""
sspicli argument probe for detect-fm-<PIN>.exe.

Attaches to detect-fm, hooks ONLY sspicli/secur32/schannel EncryptMessage and
DecryptMessage (system DLL exports — never touches the app's own module, so
an integrity check of the app's .text will not fire). For each call it emits
the six raw argument pointers plus the first two DWORDs at each one, which is
enough to identify which argument carries the PSecBufferDesc.

Waits for secur32.dll to load before resolving symbols.
"""

import frida
import sys
import time
import json
import psutil


JS = r"""
'use strict';

var encCount = 0;
var decCount = 0;

function emit(p) { try { send(p); } catch (e) {} }

function getExport(mod, sym) {
    try {
        if (typeof Module.getExportByName === 'function')
            return Module.getExportByName(mod, sym);
    } catch (e) {}
    try {
        if (typeof Module.findExportByName === 'function')
            return Module.findExportByName(mod, sym);
    } catch (e) {}
    return null;
}

function getGlobalExport(sym) {
    try {
        if (typeof Module.findGlobalExportByName === 'function')
            return Module.findGlobalExportByName(sym);
    } catch (e) {}
    try {
        if (typeof Module.findExportByName === 'function')
            return Module.findExportByName(null, sym);
    } catch (e) {}
    return null;
}

function hint(p) {
    try {
        if (!p || p.isNull()) return 'null';
        var a = p.readU32();
        var b = p.add(4).readU32();
        return '0x' + (a >>> 0).toString(16) + '/0x' + (b >>> 0).toString(16);
    } catch (e) {
        return '?';
    }
}

function preview(p) {
    try {
        if (!p || p.isNull()) return '';
        var s = p.readCString();
        if (!s) return '';
        var out = '';
        for (var i = 0; i < s.length && i < 80; i++) {
            var c = s.charCodeAt(i);
            if (c < 32 || c > 126) break;
            out += s.charAt(i);
        }
        return out;
    } catch (e) { return ''; }
}

function attachProbe(addr, where, kind) {
    Interceptor.attach(addr, {
        onEnter: function (args) {
            var n = (kind === 'enc') ? (++encCount) : (++decCount);
            if (n > 15) return;
            var row = {
                type: 'args',
                kind: kind,
                n: n,
                where: where,
                args: [],
                hints: [],
                previews: []
            };
            for (var i = 0; i < 6; i++) {
                var p = args[i];
                row.args.push(p ? p.toString() : '?');
                row.hints.push(hint(p));
                row.previews.push(preview(p));
            }
            emit(row);
        }
    });
    console.log('[*] hooked ' + where + ' @ ' + addr);
}

function hookProbe(symName, kind, fallbackOffset) {
    // 1. Try every module that could export this function.
    var mods = ['secur32.dll', 'sspicli.dll', 'schannel.dll'];
    for (var i = 0; i < mods.length; i++) {
        var a = getExport(mods[i], symName);
        if (a) {
            attachProbe(a, mods[i] + '!' + symName, kind);
            return true;
        }
    }
    // 2. Global lookup (across all loaded modules).
    var g = getGlobalExport(symName);
    if (g) {
        attachProbe(g, 'global!' + symName, kind);
        return true;
    }
    // 3. Offset fallback inside sspicli.dll.
    //    From prior v4 log:
    //      secur32!EncryptMessage -> 0x7ff878532d70, sspicli base 0x7ff878530000
    //      secur32!DecryptMessage -> 0x7ff878532cd0, sspicli base 0x7ff878530000
    try {
        var mod = null;
        var names = ['sspicli.dll', 'SSPICLI.DLL', 'Secur32.dll', 'secur32.dll'];
        for (var j = 0; j < names.length; j++) {
            try { mod = Process.getModuleByName(names[j]); if (mod) break; } catch (e) {}
        }
        if (mod) {
            var target = mod.base.add(fallbackOffset);
            attachProbe(target, mod.name + '+0x' + fallbackOffset.toString(16)
                              + ' (' + symName + ')', kind);
            return true;
        }
    } catch (e) {
        console.log('[!] offset fallback failed for ' + symName + ': ' + e);
    }
    console.log('[!] could not resolve ' + symName);
    return false;
}

function waitForModule(name, cb, tries) {
    tries = tries || 0;
    try {
        var m = Process.getModuleByName(name);
        if (m) { cb(m); return; }
    } catch (e) {}
    if (tries > 50) { console.log('[!] gave up waiting for ' + name); cb(null); return; }
    setTimeout(function () { waitForModule(name, cb, tries + 1); }, 100);
}

setImmediate(function () {
    console.log('[*] sspicli arg probe loaded (pid=' + Process.id +
                ', arch=' + Process.arch + ')');
    waitForModule('secur32.dll', function (m) {
        if (m) console.log('[*] secur32.dll present @ ' + m.base);
        hookProbe('EncryptMessage', 'enc', 0x2d70);
        hookProbe('DecryptMessage', 'dec', 0x2cd0);
        console.log('[*] probes installed');
    });
});
"""


def on_message(message, data):
    if message.get('type') != 'send':
        if message.get('type') == 'error':
            print('[frida error] ' + message.get('description', ''))
        return
    p = message.get('payload')
    if not isinstance(p, dict):
        return
    if p.get('type') == 'args':
        ts = time.strftime('%H:%M:%S')
        kind = p.get('kind', '?').upper()
        print('\n' + '=' * 78)
        print(f'[{ts}] {kind} #{p.get("n")}  {p.get("where")}')
        print('-' * 78)
        args = p.get('args', [])
        hints = p.get('hints', [''] * len(args))
        previews = p.get('previews', [''] * len(args))
        for i, a in enumerate(args):
            line = f'  arg{i}: ptr={a}  head={hints[i]}'
            if previews[i]:
                line += f'  str="{previews[i]}"'
            print(line)


def find_detect_processes():
    out = []
    for proc in psutil.process_iter(['pid', 'name']):
        try:
            name = (proc.info.get('name') or '').lower()
            if 'detect-fm' in name:
                out.append(proc.info)
        except (psutil.NoSuchProcess, psutil.AccessDenied):
            pass
    return out


def attach_and_run(pid, name):
    print(f'\n[+] Attaching to {name} (pid={pid})')
    try:
        session = frida.attach(pid)
    except Exception as e:
        print(f'[!] attach failed: {e}')
        return None
    script = session.create_script(JS)
    script.on('message', on_message)
    try:
        script.load()
    except Exception as e:
        print(f'[!] script.load failed: {e}')
        try: session.detach()
        except Exception: pass
        return None
    print(f'[+] Instrumented pid={pid}')
    return session, script


def main():
    device = frida.get_local_device()
    print(f'[*] Using device: {device}')

    existing = find_detect_processes()
    if existing:
        print('[*] Found existing processes:')
        for e in existing:
            print(f'    - {e["name"]} (pid={e["pid"]})')

    sessions = []
    for e in existing:
        r = attach_and_run(e['pid'], e['name'])
        if r:
            sessions.append(r)

    if not sessions:
        print('[*] Polling every 2s (Ctrl+C to quit)...')
        known = set()
        try:
            while True:
                for e in find_detect_processes():
                    if e['pid'] in known:
                        continue
                    r = attach_and_run(e['pid'], e['name'])
                    if r:
                        known.add(e['pid'])
                        sessions.append(r)
                time.sleep(2)
        except KeyboardInterrupt:
            pass
    else:
        print('\n[*] Monitoring. Press Ctrl+C to exit.')
        try:
            while True:
                time.sleep(1)
        except KeyboardInterrupt:
            pass

    for sess, script in sessions:
        try: script.unload()
        except Exception: pass
        try: sess.detach()
        except Exception: pass


if __name__ == '__main__':
    main()
