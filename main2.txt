#!/usr/bin/env python3
"""
sspicli argument probe for detect-fm-<PIN>.exe.

Attaches to detect-fm, hooks ONLY sspicli!EncryptMessage and
sspicli!DecryptMessage (system DLL exports — never touches the app's own
module, so an integrity check of the app's .text will not fire), and for
each call emits the six raw argument pointers plus the first two DWORDs at
each one. That's enough to identify which argument is the PSecBufferDesc.

No app-module hooks. No ws2_32. No ntdll. No backtrace. No writes to the
target's memory beyond Frida's normal trampoline on the two system exports.
"""

import frida
import sys
import time
import json
import psutil


JS = r"""
'use strict';

function emit(p) { try { send(p); } catch (e) {} }

function findExport(mod, sym) {
    try {
        if (typeof Module.getExportByName === 'function')
            return Module.getExportByName(mod, sym);
    } catch (e) {}
    try {
        if (typeof Module.findExportByName === 'function')
            return Module.findExportByName(mod, sym);
    } catch (e) {}
    return null;
}

// Read up to 8 bytes at a pointer and return as "hi/lo" hint string.
// Never throws.
function hint(ptr) {
    try {
        if (!ptr || ptr.isNull()) return 'null';
        var a = ptr.readU32();
        var b = ptr.add(4).readU32();
        return '0x' + (a >>> 0).toString(16) + '/0x' + (b >>> 0).toString(16);
    } catch (e) {
        return '?';
    }
}

// Read a short ASCII preview if the pointer looks like a C string.
function preview(ptr) {
    try {
        if (!ptr || ptr.isNull()) return '';
        var s = ptr.readCString();
        if (!s) return '';
        var out = '';
        for (var i = 0; i < s.length && i < 80; i++) {
            var c = s.charCodeAt(i);
            if (c < 32 || c > 126) break;
            out += s.charAt(i);
        }
        return out;
    } catch (e) { return ''; }
}

var encCount = 0;
var decCount = 0;

function hookProbe(modName, symName, kind) {
    var addr = findExport(modName, symName);
    if (!addr) {
        console.log('[!] ' + modName + '!' + symName + ' not found');
        return false;
    }
    try {
        Interceptor.attach(addr, {
            onEnter: function (args) {
                var n = (kind === 'enc') ? (++encCount) : (++decCount);
                // Only log first 10 of each so the terminal stays readable.
                if (n > 10) return;

                var row = {
                    type: 'args',
                    kind: kind,
                    n: n,
                    where: modName + '!' + symName,
                    args: [],
                    hints: [],
                    previews: []
                };
                for (var i = 0; i < 6; i++) {
                    var p = args[i];
                    row.args.push(p ? p.toString() : '?');
                    row.hints.push(hint(p));
                    row.previews.push(preview(p));
                }
                emit(row);
            }
        });
        console.log('[*] hooked ' + modName + '!' + symName);
        return true;
    } catch (e) {
        console.log('[!] hook ' + modName + '!' + symName + ' failed: ' + e);
        return false;
    }
}

setImmediate(function () {
    console.log('[*] sspicli arg probe loaded (pid=' + Process.id +
                ', arch=' + Process.arch + ')');
    // Prefer sspicli.dll — that is where the real code lives. Fall back to
    // secur32.dll if for some reason sspicli is not loaded.
    if (!hookProbe('sspicli.dll', 'EncryptMessage', 'enc')) {
        hookProbe('secur32.dll', 'EncryptMessage', 'enc');
    }
    if (!hookProbe('sspicli.dll', 'DecryptMessage', 'dec')) {
        hookProbe('secur32.dll', 'DecryptMessage', 'dec');
    }
    console.log('[*] probes installed');
});
"""


def on_message(message, data):
    if message.get('type') != 'send':
        if message.get('type') == 'error':
            print('[frida error] ' + message.get('description', ''))
        return
    p = message.get('payload')
    if not isinstance(p, dict):
        return
    if p.get('type') == 'args':
        ts = time.strftime('%H:%M:%S')
        kind = p.get('kind', '?').upper()
        print('\n' + '=' * 78)
        print(f'[{ts}] {kind} #{p.get("n")}  {p.get("where")}')
        print('-' * 78)
        for i, a in enumerate(p.get('args', [])):
            h = p.get('hints', [''] * 6)[i]
            pv = p.get('previews', [''] * 6)[i]
            line = f'  arg{i}: ptr={a}  head={h}'
            if pv:
                line += f'  str="{pv}"'
            print(line)


def find_detect_processes():
    out = []
    for proc in psutil.process_iter(['pid', 'name']):
        try:
            name = (proc.info.get('name') or '').lower()
            if 'detect-fm' in name:
                out.append(proc.info)
        except (psutil.NoSuchProcess, psutil.AccessDenied):
            pass
    return out


def attach_and_run(pid, name):
    print(f'\n[+] Attaching to {name} (pid={pid})')
    try:
        session = frida.attach(pid)
    except Exception as e:
        print(f'[!] attach failed: {e}')
        return None
    script = session.create_script(JS)
    script.on('message', on_message)
    try:
        script.load()
    except Exception as e:
        print(f'[!] script.load failed: {e}')
        try: session.detach()
        except Exception: pass
        return None
    print(f'[+] Instrumented pid={pid}')
    return session, script


def main():
    device = frida.get_local_device()
    print(f'[*] Using device: {device}')

    existing = find_detect_processes()
    if existing:
        print('[*] Found existing processes:')
        for e in existing:
            print(f'    - {e["name"]} (pid={e["pid"]})')

    sessions = []
    for e in existing:
        r = attach_and_run(e['pid'], e['name'])
        if r:
            sessions.append(r)

    if not sessions:
        print('[*] Polling every 2s (Ctrl+C to quit)...')
        known = set()
        try:
            while True:
                for e in find_detect_processes():
                    if e['pid'] in known:
                        continue
                    r = attach_and_run(e['pid'], e['name'])
                    if r:
                        known.add(e['pid'])
                        sessions.append(r)
                time.sleep(2)
        except KeyboardInterrupt:
            pass
    else:
        print('\n[*] Monitoring. Press Ctrl+C to exit.')
        try:
            while True:
                time.sleep(1)
        except KeyboardInterrupt:
            pass

    for sess, script in sessions:
        try: script.unload()
        except Exception: pass
        try: sess.detach()
        except Exception: pass


if __name__ == '__main__':
    main()
