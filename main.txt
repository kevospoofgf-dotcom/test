# =====================================================================
# NO-DISK LOGGING: everything goes to a bounded in-memory ring buffer.
# Nothing is ever written to Downloads or anywhere on disk.
# =====================================================================
_LOG_RING_SIZE = 5000
_LOG_RING = []
_LOG_RING_LOCK = threading.Lock()
_VERBOSE = True   # keep True for console output before hiding

def log_line(s):
    """Append to in-memory ring only. Never touches disk."""
    global _VERBOSE
    if _VERBOSE:
        try:
            print(s)
        except Exception:
            pass
    with _LOG_RING_LOCK:
        _LOG_RING.append(s)
        if len(_LOG_RING) > _LOG_RING_SIZE:
            del _LOG_RING[0:len(_LOG_RING) - _LOG_RING_SIZE]

def on_msg(m, d):
    if m["type"] != "send":
        if m["type"] == "error":
            log_line(f"[!] Frida: {m.get('description', m)}")
        return
    p = m["payload"]
    t = p.get("type")
    ts = time.strftime("%H:%M:%S")
    if t == "info":
        log_line(f"[*] {p['message']}")
    elif t == "error":
        log_line(f"[!] {p['message']}")
    elif t == "blocked":
        log_line(f"[{ts}] [BLOCKED] {p.get('api')}  {p.get('detail')}")
    else:
        log_line(f"[?] {json.dumps(p)}")

def find_existing(substr):
    out = []
    for p in psutil.process_iter(["pid", "name", "status"]):
        try:
            if substr not in (p.info["name"] or "").lower():
                continue
            if p.info["status"] in (psutil.STATUS_ZOMBIE, psutil.STATUS_DEAD):
                continue
            out.append(p.info)
        except (psutil.NoSuchProcess, psutil.AccessDenied):
            pass
    return out

def _enable_vt():
    try:
        kernel32 = ctypes.windll.kernel32
        h = kernel32.GetStdHandle(-11)
        mode = ctypes.c_uint32()
        if kernel32.GetConsoleMode(h, ctypes.byref(mode)):
            kernel32.SetConsoleMode(h, mode.value | 0x0004)
    except Exception:
        pass

def green(s):
    return f"\033[92m{s}\033[0m"

def hide_console():
    """Hide the console window entirely (so the script runs in background)."""
    try:
        kernel32 = ctypes.windll.kernel32
        user32 = ctypes.windll.user32
        hwnd = kernel32.GetConsoleWindow()
        if hwnd:
            user32.ShowWindow(hwnd, 0)  # SW_HIDE
    except Exception:
        pass

def main():
    global _VERBOSE
    _enable_vt()

    # No self-elevation, no UAC prompt, no Defender preference changes.
    # Run this from a normal (non-admin) cmd to avoid the
    # "Suspicious File Executed As Administrator Since Logon" alert.

    # First: find the target process. Keep console visible while doing so.
    log_line(f'[*] Waiting for process with "{SUBSTR}" in exe name...')
    first = None
    while not first:
        matches = find_existing(SUBSTR)
        if matches:
            first = matches[0]
            break
        time.sleep(1)

    pid = first["pid"]
    log_line(f'[*] Attaching to PID {pid} ...')

    # Attach + load guard. Any exception here = error, print in red and exit.
    try:
        session = frida.attach(pid)
        script = session.create_script(build_js())
        script.on("message", on_msg)
        script.load()
    except Exception as e:
        print(f"\033[91mError: {e}\033[0m")
        sys.exit(1)

    # Success path
    log_line("[*] Guard active.")
    _VERBOSE = False
    print()
    print(green("Works"))
    print("Press Enter to continue (run in background)...")
    try:
        input()
    except EOFError:
        pass

    # Hide console and keep running in background
    hide_console()

    attached = {pid}
    attach_failures = {}
    while True:
        try:
            time.sleep(2)
            for mm in find_existing(SUBSTR):
                np = mm["pid"]
                if np in attached:
                    continue
                last = attach_failures.get(np, 0)
                if time.time() - last < 30:
                    continue
                attach_failures[np] = time.time()
                try:
                    s2 = frida.attach(np)
                    sc2 = s2.create_script(build_js())
                    sc2.on("message", on_msg)
                    sc2.load()
                    attached.add(np)
                    attach_failures.pop(np, None)
                except Exception:
                    pass
        except KeyboardInterrupt:
            break
        except Exception:
            pass

    try:
        script.unload()
    except Exception:
        pass
    try:
        session.detach()
    except Exception:
        pass

if __name__ == "__main__":
    main()
