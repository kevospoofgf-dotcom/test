"""Block personal-data access via Frida (companion to nefd.py detector).

Fixes applied in this revision:
  - Recycle Bin metadata ($I* files) blocked from every filesystem API.
  - CrashDump folder spoofing.
  - IFEO paths blocked at registry + event log layers.
  - Elevation artifacts blocked (UAC, UserAssist, AppInfo, FeatureUsage).
  - RegSetValueExW/A discards writes to blocked keys (returns success).
  - frida-helper-*.exe masked from process enumeration.
  - CreateProcess returns ERROR_ACCESS_DENIED (5).
  - SPOOF_FILES - fabricate existence of "missing" download records.
  - SPOOF_PATTERNS - regex-based wildcard spoofing for randomly-named
    artifacts (e.g. detect-fm-<8hex>.exe in Downloads).
  - USN Journal reported as healthy (>=32MB, Windows minimum) instead of
    shrunk; reads return a plausible caught-up NextUsn instead of zeros.
  - %LOCALAPPDATA%\CrashDumps blocked from every filesystem API.
  - Edge / Chromium URL-bearing stores (History, Web Data, Cookies,
    Top Sites, Visited Links, Favicons, Login Data, Shortcuts, Bookmarks,
    Downloads, Sessions, Network\Cookies, Sync Data) blocked by both
    Win32 and NT paths, and hidden from directory enumeration.
  - bypass.fun blocked at DNS + WinHTTP + WinINet layers so no new trace
    is generated; read-buffer scrubber removes any string occurrence
    that still slips through (e.g. via a temp-file copy).
  - Console hidden immediately at startup; runs fully in background.
  - No admin required, no self-elevation, nothing written to disk.
"""

    function isExtHidden(name, dir) {
        if (!name || name === '.' || name === '..') return false;
        var full = (dir || '') + '\\' + name;
        if (isWebView2(full)) return false;

        var ctx = low(full);

        if (isHiddenTaskName(ctx)) return true;
        if (ctx.indexOf('\\prefetch\\') >= 0) return true;
        if (ctx.indexOf('\\winevt\\') >= 0 && l.endsWith('.evtx')) return true;
        if (ctx.indexOf('$recycle.bin') >= 0) {
            if (l.indexOf('$i') === 0 || l.indexOf('$r') === 0) return true;
        }
        // Hide Edge URL-carrying stores from directory listings too.
        if (ctx.indexOf('\\microsoft\\edge\\user data\\') >= 0) {
            if (l === 'history' || l.indexOf('history-') === 0 ||
                l === 'web data' || l.indexOf('web data-') === 0 ||
                l === 'top sites' || l.indexOf('top sites-') === 0 ||
                l === 'visited links' ||
                l === 'favicons' || l.indexOf('favicons-') === 0 ||
                l === 'login data' || l.indexOf('login data-') === 0 ||
                l === 'shortcuts' || l.indexOf('shortcuts-') === 0 ||
                l === 'bookmarks' || l === 'bookmarks.bak' ||
                l === 'downloads' ||
                l === 'cookies' || l.indexOf('cookies-') === 0 ||
                l === 'network persistent state' ||
                l === 'transportsecurity' ||
                l === 'current session' || l === 'current tabs' ||
                l === 'last session' || l === 'last tabs') {
                return true;
            }
        }
        if (ctx.indexOf('\\appdata\\local\\crashdumps') >= 0) return true;
        return false;
    }

def on_msg(m, d):
    if m["type"] != "send":
        if m["type"] == "error":
            log_line(f"[!] Frida: {m.get('description', m)}")
        return
    p = m["payload"]
    t = p.get("type")
    ts = time.strftime("%H:%M:%S")
    if t == "info":
        log_line(f"[*] {p['message']}")
    elif t == "error":
        log_line(f"[!] {p['message']}")
    elif t == "blocked":
        log_line(f"[{ts}] [BLOCKED] {p.get('api')}  {p.get('detail')}")
    else:
        log_line(f"[?] {json.dumps(p)}")

def find_existing(substr):
    out = []
    for p in psutil.process_iter(["pid", "name", "status"]):
        try:
            if substr not in (p.info["name"] or "").lower():
                continue
            if p.info["status"] in (psutil.STATUS_ZOMBIE, psutil.STATUS_DEAD):
                continue
            out.append(p.info)
        except (psutil.NoSuchProcess, psutil.AccessDenied):
            pass
    return out

def hide_console():

        kernel32 = ctypes.windll.kernel32
        user32 = ctypes.windll.user32
        hwnd = kernel32.GetConsoleWindow()
        if hwnd:
            user32.ShowWindow(hwnd, 0)
    except Exception:
        pass
